### `terraform/main.tf`

```hcl
# ──────────────────────────────────────────────
# Terraform & Provider Configuration
# ──────────────────────────────────────────────

terraform {
  required_version = ">= 0.14.0"
  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53.0"
    }
  }
}

provider "openstack" {
  user_name   = var.openstack_user_name
  tenant_name = var.openstack_tenant_name
  password    = var.openstack_password
  auth_url    = var.openstack_auth_url
  region      = var.openstack_region
}

# ──────────────────────────────────────────────
# Security Group
# ──────────────────────────────────────────────

resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = var.security_group_name
  description = "Security group for web server - allows SSH and HTTP"
}

resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = var.allowed_ssh_cidr
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = "Allow inbound SSH"
}

resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = var.allowed_http_cidr
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = "Allow inbound HTTP"
}

# ──────────────────────────────────────────────
# Floating IP
# ──────────────────────────────────────────────

resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.floating_ip_pool
}

# ──────────────────────────────────────────────
# Compute Instance
# ──────────────────────────────────────────────

resource "openstack_compute_instance_v2" "web_server" {
  name            = var.instance_name
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.ssh_key_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  network {
    name = var.network_name
  }
}

# ──────────────────────────────────────────────
# Associate Floating IP to Instance
# ──────────────────────────────────────────────

resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}

# ──────────────────────────────────────────────
# Generate Ansible Inventory
# ──────────────────────────────────────────────

resource "local_file" "ansible_inventory" {
  content = templatefile("${path.module}/../ansible/inventory.ini.tpl", {
    web_server_ip   = openstack_networking_floatingip_v2.web_fip.address
    ssh_user        = var.ssh_user
    private_key_path = var.private_key_path
  })
  filename = "${path.module}/../ansible/inventory.ini"

  depends_on = [openstack_compute_floatingip_associate_v2.web_fip_assoc]
}
```

### `ansible/playbook.yml`

```yaml
---
# ──────────────────────────────────────────────
# Playbook: Configure Nginx Web Server
# ──────────────────────────────────────────────

- name: Configure web server with Nginx
  hosts: webservers
  become: true
  gather_facts: true

  vars_files:
    - group_vars/all.yml

  pre_tasks:
    - name: Display deployment information
      ansible.builtin.debug:
        msg:
          - '──────────────────────────────────────'
          - '  Nginx Web Server Configuration'
          - '──────────────────────────────────────'
          - '  Target host : {{ inventory_hostname }}'
          - '  Listen port : {{ nginx_listen_port }}'
          - '  Document root: {{ nginx_document_root }}'
          - '──────────────────────────────────────'

    - name: Wait for the instance to become reachable via SSH
      ansible.builtin.wait_for_connection:
        delay: 10
        timeout: 300

    - name: Gather facts after connection is established
      ansible.builtin.setup:

  tasks:
    # ── Package Installation ──────────────────

    - name: Update apt cache
      ansible.builtin.apt:
        update_cache: true
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"

    - name: Install Nginx (Debian/Ubuntu)
      ansible.builtin.apt:
        name: nginx
        state: present
      when: ansible_os_family == "Debian"

    - name: Install Nginx (RedHat/CentOS)
      ansible.builtin.yum:
        name: nginx
        state: present
      when: ansible_os_family == "RedHat"

    # ── Directory Setup ───────────────────────

    - name: Ensure document root directory exists
      ansible.builtin.file:
        path: '{{ nginx_document_root }}'
        state: directory
        owner: www-data
        group: www-data
        mode: '0755'

    # ── Deploy Static HTML ────────────────────

    - name: Deploy static index.html
      ansible.builtin.template:
        src: templates/index.html.j2
        dest: '{{ nginx_document_root }}/index.html'
        owner: www-data
        group: www-data
        mode: '0644'
      notify: Restart Nginx

    # ── Nginx Configuration ───────────────────

    - name: Remove default Nginx site configuration
      ansible.builtin.file:
        path: /etc/nginx/sites-enabled/default
        state: absent
      notify: Restart Nginx
      when: ansible_os_family == "Debian"

    - name: Deploy Nginx server block configuration
      ansible.builtin.template:
        src: templates/nginx.conf.j2
        dest: /etc/nginx/sites-available/webserver.conf
        owner: root
        group: root
        mode: '0644'
      notify: Restart Nginx
      when: ansible_os_family == "Debian"

    - name: Enable Nginx server block (symlink)
      ansible.builtin.file:
        src: /etc/nginx/sites-available/webserver.conf
        dest: /etc/nginx/sites-enabled/webserver.conf
        state: link
      notify: Restart Nginx
      when: ansible_os_family == "Debian"

    - name: Validate Nginx configuration
      ansible.builtin.command: nginx -t
      changed_when: false
      register: nginx_test

    - name: Display Nginx config test result
      ansible.builtin.debug:
        msg: '{{ nginx_test.stderr_lines }}'

    # ── Service Management ────────────────────

    - name: Enable and start Nginx service
      ansible.builtin.systemd:
        name: '{{ nginx_service_name }}'
        enabled: '{{ nginx_service_enabled }}'
        state: '{{ nginx_service_state }}'

  handlers:
    - name: Restart Nginx
      ansible.builtin.systemd:
        name: '{{ nginx_service_name }}'
        state: restarted

  post_tasks:
    - name: Verify Nginx is responding on port {{ nginx_listen_port }}
      ansible.builtin.uri:
        url: 'http://localhost:{{ nginx_listen_port }}'
        status_code: 200
        return_content: false
      register: health_check
      retries: 3
      delay: 5
      until: health_check.status == 200

    - name: Deployment summary
      ansible.builtin.debug:
        msg:
          - '──────────────────────────────────────'
          - '  Deployment Complete!'
          - '──────────────────────────────────────'
          - '  Nginx status : Running'
          - '  Health check : HTTP {{ health_check.status }}'
          - '  Access URL   : http://{{ ansible_host }}:{{ nginx_listen_port }}'
          - '──────────────────────────────────────'
```

## Deployment Instructions

### Prerequisites

| Tool      | Minimum Version |
| --------- | --------------- |
| Terraform | >= 0.14.0       |
| Ansible   | >= 2.14         |
| Python    | >= 3.8          |

Ensure your OpenStack SSH keypair is already registered and you have the corresponding private key locally.

### Step 1 — Configure Terraform Variables

```bash
cd terraform/
cp terraform.tfvars.example terraform.tfvars
```

Edit `terraform.tfvars` with your OpenStack credentials and environment-specific values. **Alternatively**, export environment variables so credentials never touch disk:

```bash
export TF_VAR_openstack_auth_url="http://controller:5000/v3"
export TF_VAR_openstack_user_name="your-username"
export TF_VAR_openstack_password="your-password"
export TF_VAR_openstack_tenant_name="your-project"
export TF_VAR_openstack_region="RegionOne"
export TF_VAR_ssh_key_name="my-keypair"
export TF_VAR_private_key_path="$HOME/.ssh/id_rsa"
export TF_VAR_network_name="private-net"
```

### Step 2 — Provision Infrastructure with Terraform

```bash
# Initialize providers and modules
terraform init

# Preview what will be created
terraform plan

# Apply — type 'yes' when prompted
terraform apply
```

After `apply` completes, Terraform outputs the floating IP, web URL, and SSH command. It also generates `ansible/inventory.ini` automatically.

### Step 3 — Configure the Server with Ansible

```bash
cd ../ansible/

# (Optional) Verify connectivity first
ansible -i inventory.ini webservers -m ping

# Run the playbook
ansible-playbook -i inventory.ini playbook.yml
```

### Step 4 — Verify HTTP Access

**From a browser:**

Open the URL shown in the Terraform output (`web_url`):

```
http://<floating-ip>
```

**From the command line:**

```bash
# Using the floating IP from Terraform output
curl http://$(cd ../terraform && terraform output -raw floating_ip)
```

Expected result: an HTML page displaying **"Hello from OpenStack!"** with the server hostname and IP address.

**Verify via SSH:**

```bash
# Use the SSH command from Terraform output
ssh -i ~/.ssh/id_rsa ubuntu@<floating-ip>

# On the server, verify Nginx is running
sudo systemctl status nginx
curl -s localhost
```

### Step 5 — Tear Down (When Finished)

```bash
cd ../terraform/
terraform destroy
```

Type `yes` to confirm. This removes the instance, security group, floating IP, and all associated resources.
