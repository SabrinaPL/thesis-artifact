### `terraform/main.tf`

```hcl
# ─── Provider Configuration ─────────────────────────────────────────────────
terraform {
  required_version = ">= 0.14.0"
  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53.0"
    }
  }
}

provider "openstack" {
  user_name   = var.user_name
  tenant_name = var.tenant_name
  password    = var.password
  auth_url    = var.auth_url
  region      = var.region
}

# ─── Security Group ─────────────────────────────────────────────────────────
resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = "${var.instance_name}-sg"
  description = "Security group allowing SSH and HTTP inbound traffic"
}

resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# ─── Floating IP ────────────────────────────────────────────────────────────
resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.external_network_name
}

# ─── Compute Instance ───────────────────────────────────────────────────────
resource "openstack_compute_instance_v2" "web_server" {
  name            = var.instance_name
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.ssh_key_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  network {
    name = var.network_name
  }
}

# ─── Associate Floating IP to Instance ──────────────────────────────────────
resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}

# ─── Generate Ansible Inventory ─────────────────────────────────────────────
resource "local_file" "ansible_inventory" {
  content = templatefile("${path.module}/../ansible/inventory.ini.tpl", {
    server_ip        = openstack_networking_floatingip_v2.web_fip.address
    ssh_user         = var.ssh_user
    private_key_path = var.private_key_path
  })
  filename = "${path.module}/../ansible/inventory.ini"

  depends_on = [openstack_compute_floatingip_associate_v2.web_fip_assoc]
}
```

### `ansible/playbook.yml`

```yaml
---
- name: Configure web server with Nginx
  hosts: webservers
  become: true
  gather_facts: true

  vars:
    nginx_port: 80
    document_root: /var/www/html
    index_file: index.html
    nginx_service_name: nginx

  tasks:
    # ── 1. Wait for the instance to become reachable via SSH ──────────────
    - name: Wait for SSH connectivity
      ansible.builtin.wait_for_connection:
        delay: 10
        timeout: 300

    # ── 2. Update apt cache (Debian/Ubuntu) ───────────────────────────────
    - name: Update apt package cache
      ansible.builtin.apt:
        update_cache: yes
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"

    # ── 3. Install Nginx via system package manager ───────────────────────
    - name: Install Nginx package
      ansible.builtin.package:
        name: '{{ nginx_service_name }}'
        state: present

    # ── 4. Ensure document root directory exists ──────────────────────────
    - name: Ensure document root exists
      ansible.builtin.file:
        path: '{{ document_root }}'
        state: directory
        owner: www-data
        group: www-data
        mode: '0755'

    # ── 5. Deploy static HTML file ───────────────────────────────────────
    - name: Deploy static index.html
      ansible.builtin.copy:
        src: files/{{ index_file }}
        dest: '{{ document_root }}/{{ index_file }}'
        owner: www-data
        group: www-data
        mode: '0644'
      notify: Reload Nginx

    # ── 6. Configure Nginx to listen on the specified port ────────────────
    - name: Configure Nginx default site
      ansible.builtin.template:
        src: templates/default_site.conf.j2
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: '0644'
      notify: Reload Nginx

    # ── 7. Enable and start Nginx service ─────────────────────────────────
    - name: Ensure Nginx is enabled and running
      ansible.builtin.service:
        name: '{{ nginx_service_name }}'
        state: started
        enabled: true

  handlers:
    - name: Reload Nginx
      ansible.builtin.service:
        name: '{{ nginx_service_name }}'
        state: reloaded
```

### `ansible/templates/default_site.conf.j2`

server {

    root {{ document_root }};

}

## Deployment Instructions

### Prerequisites

| Requirement                         | Minimum Version |
| ----------------------------------- | --------------- |
| Terraform                           | >= 0.14.0       |
| Ansible                             | >= 2.14         |
| Python                              | >= 3.8          |
| SSH keypair registered in OpenStack | —               |

### Step 1: Configure Terraform Variables

```bash
cd terraform/

# Create your variable file from the example
cp terraform.tfvars.example terraform.tfvars

# Edit with your actual OpenStack credentials and configuration
# IMPORTANT: Never commit terraform.tfvars to version control
nano terraform.tfvars
```

Fill in all values—authentication URL, project name, credentials, image name, flavor, network, and your SSH keypair details.

### Step 2: Initialize and Apply Terraform

```bash
# Initialize Terraform — downloads the OpenStack provider
terraform init

# Preview all resources that will be created
terraform plan

# Provision the infrastructure (type 'yes' when prompted)
terraform apply
```

**Expected output after apply:**

```
Outputs:

floating_ip     = "192.168.1.100"
instance_name   = "web-server"
security_group  = "web-server-sg"
ssh_command      = "ssh -i ~/.ssh/id_rsa ubuntu@192.168.1.100"
web_url          = "http://192.168.1.100"
```

Terraform automatically generates the Ansible inventory file at `ansible/inventory.ini`.

### Step 3: Run the Ansible Playbook

```bash
cd ../ansible/

# Run the playbook to install and configure Nginx
ansible-playbook -i inventory.ini playbook.yml
```

The playbook will:

1. Wait for SSH connectivity to the new instance
2. Update the package cache
3. Install Nginx via the system package manager (`apt`/`yum`)
4. Create the document root directory
5. Deploy the static HTML page to `/var/www/html/index.html`
6. Configure Nginx to listen on port 80
7. Enable and start the Nginx service

### Step 4: Verify HTTP Access

```bash
# Retrieve the floating IP from Terraform outputs
cd ../terraform/
export WEB_IP=$(terraform output -raw floating_ip)

# Test via curl
curl http://${WEB_IP}

# Or open in a browser
echo "Open in browser: http://${WEB_IP}"
```

You should see the styled HTML page reading **"Hello from OpenStack!"**

### Step 5: Verify SSH Access (Optional)

```bash
# Use the SSH command from Terraform output
terraform output -raw ssh_command
# Then run the outputted command, e.g.:
# ssh -i ~/.ssh/id_rsa ubuntu@192.168.1.100
```

### Step 6: Tear Down (When Finished)

```bash
cd terraform/

# Destroy all provisioned infrastructure
terraform destroy
```

Type `yes` when prompted. This removes the instance, security group, floating IP, and all associated resources.

## Security Notes

| Concern                           | Mitigation                                                                     |
| --------------------------------- | ------------------------------------------------------------------------------ |
| Credentials in `terraform.tfvars` | Add `*.tfvars` to `.gitignore`; alternatively use `OS_*` environment variables |
| SSH key path                      | Passed as a variable, never hardcoded                                          |
| State file contains secrets       | Use a remote backend (e.g., Swift, S3) with encryption in production           |
| SSH open to `0.0.0.0/0`           | Restrict `remote_ip_prefix` to your IP range for production use                |

### Using Environment Variables Instead of `terraform.tfvars`

```bash
export OS_AUTH_URL="http://your-controller:5000/v3"
export OS_TENANT_NAME="my-project"
export OS_USERNAME="my-username"
export OS_PASSWORD="my-password"
export OS_REGION_NAME="RegionOne"

# Then set only the non-sensitive variables in terraform.tfvars
# and remove auth_url, tenant_name, user_name, password from it
```

When `OS_*` environment variables are set, the OpenStack provider reads them automatically, so you can omit them from the provider block and the tfvars file entirely.
