### `terraform/main.tf`

```hcl
# ──────────────────────────────────────────────
# Provider & version constraints
# ──────────────────────────────────────────────

terraform {
  required_version = ">= 0.14.0"
  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53.0"
    }
  }
}

provider "openstack" {
  user_name   = var.openstack_user_name
  tenant_name = var.openstack_tenant_name
  password    = var.openstack_password
  auth_url    = var.openstack_auth_url
  region      = var.openstack_region
}

# ──────────────────────────────────────────────
# Security Group — SSH + HTTP ingress
# ──────────────────────────────────────────────

resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = "${var.instance_name}-sg"
  description = "Allow inbound SSH (22) and HTTP (80)"
}

resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# ──────────────────────────────────────────────
# Floating IP (public address)
# ──────────────────────────────────────────────

resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.floating_ip_pool
}

# ──────────────────────────────────────────────
# Compute Instance
# ──────────────────────────────────────────────

resource "openstack_compute_instance_v2" "web_server" {
  name            = var.instance_name
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.ssh_key_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  network {
    name = var.network_name
  }
}

# ──────────────────────────────────────────────
# Associate Floating IP with Instance
# ──────────────────────────────────────────────

resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}

# ──────────────────────────────────────────────
# Generate Ansible Inventory
# ──────────────────────────────────────────────

resource "local_file" "ansible_inventory" {
  filename = "${path.module}/../ansible/inventory.ini"
  content  = <<-EOF
    [webservers]
    ${openstack_networking_floatingip_v2.web_fip.address} ansible_user=${var.ssh_user} ansible_ssh_private_key_file=${var.ssh_private_key_path}
  EOF

  depends_on = [openstack_compute_floatingip_associate_v2.web_fip_assoc]
}
```

### `ansible/playbook.yml`

```yaml
---
# ──────────────────────────────────────────────
# Web Server Configuration Playbook
# ──────────────────────────────────────────────
- name: Configure Nginx Web Server
  hosts: webservers
  become: true
  gather_facts: true

  vars:
    nginx_listen_port: 80
    document_root: /var/www/html
    index_file: index.html
    server_name: '{{ ansible_hostname }}'
    page_title: 'Welcome'
    page_heading: 'Hello from {{ ansible_hostname }}'
    page_message: 'This server was provisioned with Terraform and configured with Ansible.'

  tasks:
    # ── Package Installation ──────────────────
    - name: Update apt cache (Debian/Ubuntu)
      ansible.builtin.apt:
        update_cache: true
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"

    - name: Install nginx
      ansible.builtin.package:
        name: nginx
        state: present

    # ── Document Root ─────────────────────────
    - name: Ensure document root exists
      ansible.builtin.file:
        path: '{{ document_root }}'
        state: directory
        owner: root
        group: root
        mode: '0755'

    # ── Static HTML Page ──────────────────────
    - name: Deploy static index.html
      ansible.builtin.template:
        src: templates/index.html.j2
        dest: '{{ document_root }}/{{ index_file }}'
        owner: root
        group: root
        mode: '0644'
      notify: Reload nginx

    # ── Nginx Default Site Configuration ──────
    - name: Configure nginx default site
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: '0644'
        content: |
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              server_name {{ server_name }};

              root {{ document_root }};
              index {{ index_file }};

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      notify: Reload nginx
      when: ansible_os_family == "Debian"

    # ── Service Management ────────────────────
    - name: Ensure nginx is enabled and running
      ansible.builtin.service:
        name: nginx
        state: started
        enabled: true

  handlers:
    - name: Reload nginx
      ansible.builtin.service:
        name: nginx
        state: reloaded
```

## Deployment Instructions

### Prerequisites

| Tool                                | Minimum Version |
| ----------------------------------- | --------------- |
| Terraform                           | >= 0.14         |
| Ansible                             | >= 2.14         |
| Python                              | >= 3.8          |
| OpenStack account with API access   | —               |
| SSH keypair registered in OpenStack | —               |

### Step 1: Configure Terraform Variables

```bash
cd terraform/
cp terraform.tfvars.example terraform.tfvars
```

Edit `terraform.tfvars` with your actual OpenStack credentials, keypair name, private key path, network name, and image/flavor choices. **Never commit `terraform.tfvars` to version control** — add it to `.gitignore`.

Alternatively, export credentials as environment variables to avoid storing them in files:

```bash
export OS_AUTH_URL="http://your-keystone-host:5000/v3"
export OS_USERNAME="your-username"
export OS_PASSWORD="your-password"
export OS_TENANT_NAME="your-project"
export OS_REGION_NAME="RegionOne"
```

If using environment variables, you can remove the provider credential arguments from `main.tf` and the corresponding variable declarations, since the OpenStack provider reads `OS_*` variables automatically.

### Step 2: Provision Infrastructure with Terraform

```bash
# Initialize — downloads the OpenStack provider plugin
terraform init

# Preview the execution plan
terraform plan

# Apply — provisions the VM, security group, and floating IP
terraform apply
```

Type `yes` when prompted. On success, Terraform prints outputs including the floating IP and generates the Ansible inventory at `ansible/inventory.ini`.

### Step 3: Wait for Instance to Boot

Give the instance 30–60 seconds to finish booting and accept SSH connections:

```bash
# Use the floating IP from Terraform output
FLOATING_IP=$(terraform output -raw floating_ip)

# Test SSH connectivity
ssh -i <your-private-key-path> -o ConnectTimeout=10 ubuntu@${FLOATING_IP} echo "SSH OK"
```

### Step 4: Configure the Server with Ansible

```bash
cd ../ansible/

# Verify inventory was generated
cat inventory.ini

# Run the playbook
ansible-playbook -i inventory.ini playbook.yml
```

If the SSH host key is not yet trusted, either add it manually or set the environment variable:

```bash
export ANSIBLE_HOST_KEY_CHECKING=False
ansible-playbook -i inventory.ini playbook.yml
```

### Step 5: Verify HTTP Access

```bash
# From Terraform outputs:
FLOATING_IP=$(cd ../terraform && terraform output -raw floating_ip)

# Verify with curl
curl http://${FLOATING_IP}

# Or open in a browser
echo "Open in browser: http://${FLOATING_IP}"
```

You should see the styled HTML page with the hostname and OS details rendered by the Jinja2 template.

### Step 6: Tear Down (when finished)

```bash
cd ../terraform/
terraform destroy
```

Type `yes` to confirm. This removes the compute instance, security group, floating IP, and the generated inventory file.

## Security Notes

| Concern                           | Mitigation                                                             |
| --------------------------------- | ---------------------------------------------------------------------- |
| Credentials in `terraform.tfvars` | Add to `.gitignore`; prefer `OS_*` env vars or a secrets manager       |
| SSH open to `0.0.0.0/0`           | Restrict `remote_ip_prefix` in the security group rule to your IP/CIDR |
| Sensitive Terraform variables     | Marked `sensitive = true` so they are redacted in CLI output           |
| SSH host key checking             | Re-enable after initial deployment for production use                  |
