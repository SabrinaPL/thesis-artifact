### `terraform/main.tf`

```hcl
# -------------------------------------------------------
# Terraform Configuration for OpenStack Web Server
#
# Provisions:
#   - A security group allowing SSH (22) and HTTP (80)
#   - A compute instance with the specified image/flavor
#   - A floating IP for external access
#
# Authentication is handled via environment variables
# (OS_AUTH_URL, OS_USERNAME, OS_PASSWORD, etc.) or
# via a clouds.yaml reference. No credentials are
# hardcoded in this file.
# -------------------------------------------------------

terraform {
  required_version = ">= 0.14.0"
  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53.0"
    }
  }
}

# The provider block is intentionally left empty so that
# authentication is sourced from environment variables
# (e.g., OS_AUTH_URL, OS_USERNAME, OS_PASSWORD, OS_TENANT_NAME)
# or from a clouds.yaml file via the OS_CLOUD env var.
provider "openstack" {
  auth_url = var.auth_url != "" ? var.auth_url : null
  cloud    = var.cloud_name != "" ? var.cloud_name : null
}

# -------------------------------------------------------
# Security Group
# -------------------------------------------------------
resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = var.security_group_name
  description = "Security group for web server - allows SSH and HTTP inbound"
}

# Rule: Allow inbound SSH (port 22) from any source
resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# Rule: Allow inbound HTTP (port 80) from any source
resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# -------------------------------------------------------
# Floating IP (for external/public access)
# -------------------------------------------------------
resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.floating_ip_pool
}

# -------------------------------------------------------
# Compute Instance
# -------------------------------------------------------
resource "openstack_compute_instance_v2" "web_server" {
  name            = var.instance_name
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.ssh_key_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  network {
    name = var.network_name
  }
}

# -------------------------------------------------------
# Associate Floating IP with Instance
# -------------------------------------------------------
resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}

# -------------------------------------------------------
# Generate Ansible Inventory
# -------------------------------------------------------
resource "local_file" "ansible_inventory" {
  content = templatefile("${path.module}/../ansible/inventory.ini.tpl", {
    server_ip        = openstack_networking_floatingip_v2.web_fip.address
    ssh_user         = var.ssh_user
    private_key_path = var.private_key_path
  })
  filename = "${path.module}/../ansible/inventory.ini"

  depends_on = [openstack_compute_floatingip_associate_v2.web_fip_assoc]
}
```

### `ansible/playbook.yml`

```yaml
---
# -------------------------------------------------------
# Ansible Playbook: Configure Nginx Web Server
#
# This playbook:
#   1. Installs nginx via the system package manager
#   2. Deploys a static HTML page to /var/www/html/
#   3. Configures nginx to listen on port 80
#   4. Ensures nginx is running and enabled at boot
#
# All configurable values are expressed as variables
# with sensible defaults that can be overridden.
# -------------------------------------------------------

- name: Configure Nginx Web Server
  hosts: webservers
  become: true
  gather_facts: true

  vars:
    # Document root where the static site is served from
    nginx_document_root: "/var/www/html"

    # Filename for the deployed HTML page
    nginx_index_file: "index.html"

    # Port on which nginx listens
    nginx_listen_port: 80

    # Name of the nginx package (overridable per distro)
    nginx_package_name: "nginx"

    # Name of the nginx service
    nginx_service_name: "nginx"

    # Server name for the nginx virtual host
    nginx_server_name: "_"

  tasks:
    # ---- Package Installation ----

    - name: Update apt cache (Debian/Ubuntu)
      ansible.builtin.apt:
        update_cache: true
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"

    - name: Install nginx
      ansible.builtin.package:
        name: "{{ nginx_package_name }}"
        state: present

    # ---- Document Root ----

    - name: Ensure document root directory exists
      ansible.builtin.file:
        path: "{{ nginx_document_root }}"
        state: directory
        owner: root
        group: root
        mode: "0755"

    # ---- Deploy Static Content ----

    - name: Deploy static HTML file
      ansible.builtin.copy:
        src: "files/index.html"
        dest: "{{ nginx_document_root }}/{{ nginx_index_file }}"
        owner: root
        group: root
        mode: "0644"
      notify: Reload nginx

    # ---- Nginx Configuration ----

    - name: Configure nginx default site
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: "0644"
        content: |
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              server_name {{ nginx_server_name }};

              root {{ nginx_document_root }};
              index {{ nginx_index_file }};

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      notify: Reload nginx
      when: ansible_os_family == "Debian"

    - name: Configure nginx (RHEL/CentOS)
      ansible.builtin.copy:
        dest: /etc/nginx/conf.d/default.conf
        owner: root
        group: root
        mode: "0644"
        content: |
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              server_name {{ nginx_server_name }};

              root {{ nginx_document_root }};
              index {{ nginx_index_file }};

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      notify: Reload nginx
      when: ansible_os_family == "RedHat"

    # ---- Service Management ----

    - name: Enable and start nginx service
      ansible.builtin.systemd:
        name: "{{ nginx_service_name }}"
        state: started
        enabled: true

  handlers:
    - name: Reload nginx
      ansible.builtin.systemd:
        name: "{{ nginx_service_name }}"
        state: reloaded
```

## Deployment Instructions

### Prerequisites

| Tool | Minimum Version | Purpose |
|------|----------------|---------|
| Terraform | >= 0.14 | Infrastructure provisioning |
| Ansible | >= 2.14 | Configuration management |
| Python | >= 3.8 | Ansible runtime |
| OpenStack CLI (optional) | any | Credential verification |

### Step 1: Configure OpenStack Authentication

Set environment variables for OpenStack access. **Never hardcode credentials in configuration files.**

```bash
# Source your OpenStack RC file (downloaded from Horizon dashboard)
source ~/openrc.sh

# Or manually export the required variables:
export OS_AUTH_URL="https://your-keystone-endpoint:5000/v3"
export OS_USERNAME="your-username"
export OS_PASSWORD="your-password"
export OS_PROJECT_NAME="your-project"
export OS_USER_DOMAIN_NAME="Default"
export OS_PROJECT_DOMAIN_NAME="Default"
export OS_REGION_NAME="RegionOne"
```

Alternatively, if using `clouds.yaml`:

```bash
export OS_CLOUD="your-cloud-name"
```

### Step 2: Prepare Terraform Variables

```bash
cd terraform/

# Copy the example and edit with your values
cp terraform.tfvars.example terraform.tfvars
```

Edit `terraform.tfvars` to match your environment:

```hcl
ssh_key_name     = "my-keypair"          # Must already exist in OpenStack
private_key_path = "~/.ssh/id_rsa"       # Matching private key on your machine
image_name       = "Ubuntu-22.04"        # Available image in your OpenStack
flavor_name      = "m1.small"            # Available flavor
network_name     = "private-net"         # Your network name
floating_ip_pool = "external-net"        # Your floating IP pool name
ssh_user         = "ubuntu"              # Default user for your image
```

### Step 3: Provision Infrastructure with Terraform

```bash
# Initialize Terraform (downloads the OpenStack provider)
terraform init

# Preview the execution plan
terraform plan

# Apply the plan to create resources
terraform apply
```

When prompted, type `yes` to confirm. Upon completion, Terraform will display outputs:

```
Outputs:

floating_ip  = "192.168.1.100"
instance_name = "web-server"
web_url      = "http://192.168.1.100"
ssh_command  = "ssh -i ~/.ssh/id_rsa ubuntu@192.168.1.100"
```

Terraform also generates the Ansible inventory file at `ansible/inventory.ini`.

### Step 4: Configure the Server with Ansible

```bash
cd ../ansible/

# Wait briefly for the instance to finish booting and become SSH-accessible
sleep 30

# Verify connectivity
ansible webservers -i inventory.ini -m ping

# Run the playbook
ansible-playbook -i inventory.ini playbook.yml
```

Expected output (abbreviated):

```

### Step 5: Verify HTTP Access

```bash
# Get the floating IP from Terraform outputs
cd ../terraform/
FLOATING_IP=$(terraform output -raw floating_ip)

# Test with curl
curl http://${FLOATING_IP}
```

You should see the HTML content from `index.html`. You can also open `http://<floating_ip>` in a web browser to see the styled page.

To verify via SSH:

```bash
# SSH into the instance
ssh -i ~/.ssh/id_rsa ubuntu@${FLOATING_IP}

# On the instance, verify nginx is running
sudo systemctl status nginx
curl localhost
```

### Step 6: Tear Down (When Finished)

```bash
cd terraform/
terraform destroy
```

Type `yes` when prompted. This removes the compute instance, floating IP, and security group from OpenStack.

## Summary of Design Decisions

| Decision | Rationale |
|----------|-----------|
| **No hardcoded credentials** | OpenStack auth sourced from environment variables or `clouds.yaml` |
| **All values as variables** | Every configurable parameter (image, flavor, network, ports, paths) is a Terraform or Ansible variable with a default |
| **Template-generated inventory** | Terraform automatically produces the Ansible inventory, eliminating manual IP entry |
| **Idempotent playbook** | Ansible's `package`, `copy`, `file`, and `systemd` modules are idempotent — safe to rerun |
| **OS-family-aware configuration** | Nginx config task branches for Debian vs. RedHat families |
| **Handler-based reload** | Nginx only reloads when configuration actually changes, avoiding unnecessary restarts |
```
