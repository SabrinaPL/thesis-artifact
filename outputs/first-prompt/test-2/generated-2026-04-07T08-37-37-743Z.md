### `terraform/main.tf`
```hcl
provider "openstack" {
  auth_url           = var.auth_url
  region             = var.region
  user_name          = var.user_name
  password           = var.password
  tenant_name        = var.project_name
  user_domain_name   = var.user_domain_name
  project_domain_name = var.project_domain_name
}

# Security group with inbound SSH and HTTP
resource "openstack_networking_secgroup_v2" "web" {
  name        = var.security_group_name
  description = "Allow SSH(22) and HTTP(80)"
}

resource "openstack_networking_secgroup_rule_v2" "ingress_ssh" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web.id
}

resource "openstack_networking_secgroup_rule_v2" "ingress_http" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web.id
}

data "openstack_networking_network_v2" "net" {
  name = var.network_name
}

# VM instance
resource "openstack_compute_instance_v2" "web" {
  name        = var.instance_name
  image_name  = var.image_name
  flavor_name = var.flavor_name
  key_pair    = var.ssh_key_name

  network {
    uuid = data.openstack_networking_network_v2.net.id
  }

  security_groups = [openstack_networking_secgroup_v2.web.name]
}

# Optional floating IP for easy HTTP access
data "openstack_networking_network_v2" "external" {
  count = var.allocate_floating_ip ? 1 : 0
  name  = var.external_network_name
}

resource "openstack_networking_floatingip_v2" "fip" {
  count = var.allocate_floating_ip ? 1 : 0
  pool  = data.openstack_networking_network_v2.external[0].name
}

resource "openstack_compute_floatingip_associate_v2" "assoc" {
  count       = var.allocate_floating_ip ? 1 : 0
  floating_ip = openstack_networking_floatingip_v2.fip[0].address
  instance_id = openstack_compute_instance_v2.web.id
}
```

## Ansible — install and configure nginx + static index.html

### `ansible/site.yml`
```yaml
---
- name: Configure web server with nginx
  hosts: web
  become: true
  vars:
    web_root: /var/www/html
    index_file: "{{ web_root }}/index.html"
    site_title: "Hello from OpenStack + Terraform + Ansible"
    listen_port: 80

  tasks:
    - name: Install nginx (Debian/Ubuntu)
      ansible.builtin.apt:
        name: nginx
        state: present
        update_cache: true
      when: ansible_facts.os_family == "Debian"

    - name: Install nginx (RHEL/Rocky/CentOS)
      ansible.builtin.yum:
        name: nginx
        state: present
      when: ansible_facts.os_family == "RedHat"

    - name: Ensure nginx is enabled and running
      ansible.builtin.service:
        name: nginx
        state: started
        enabled: true

    - name: Ensure web root exists
      ansible.builtin.file:
        path: "{{ web_root }}"
        state: directory
        mode: "0755"

    - name: Deploy static index.html
      ansible.builtin.copy:
        dest: "{{ index_file }}"
        mode: "0644"
        content: |
          <!doctype html>
          <html>
            <head>
              <meta charset="utf-8" />
              <title>{{ site_title }}</title>
            </head>
            <body>
              <h1>{{ site_title }}</h1>
              <p>Served by nginx on port {{ listen_port }}.</p>
            </body>
          </html>

    # Ensure nginx listens on port 80 (default), but keep it explicit:
    - name: Ensure default site listens on port 80 (Debian/Ubuntu)
      ansible.builtin.lineinfile:
        path: /etc/nginx/sites-available/default
        regexp: '^\s*listen\s+'
        line: "        listen {{ listen_port }};"
      notify: Reload nginx
      when: ansible_facts.os_family == "Debian"

  handlers:
    - name: Reload nginx
      ansible.builtin.service:
        name: nginx
        state: reloaded
```

## How to run (deploy + configure + verify)

### 1) Terraform apply
```bash
cd iac/terraform
cp terraform.tfvars.example terraform.tfvars
# edit terraform.tfvars with your real values (don’t commit it)

terraform init
terraform plan
terraform apply
```

Get the server IP:
```bash
terraform output -raw public_ip
```
(If you set `allocate_floating_ip=false`, you’ll need to use a reachable fixed IP and routing/VPN/bastion as appropriate.)

### 2) Build Ansible inventory from Terraform outputs
From `iac/terraform`:
```bash
WEB_HOST="$(terraform output -raw public_ip)"
SSH_USER="$(terraform output -raw ssh_user)"
PRIVATE_KEY_PATH="$(terraform output -raw private_key_path)"

cd ../ansible
sed \
  -e "s|\${WEB_HOST}|$WEB_HOST|g" \
  -e "s|\${SSH_USER}|$SSH_USER|g" \
  -e "s|\${PRIVATE_KEY_PATH}|$PRIVATE_KEY_PATH|g" \
  inventory.ini.tmpl > inventory.ini
```

### 3) Run Ansible
```bash
ansible-playbook -i inventory.ini site.yml
```

### 4) Verify via HTTP
```bash
curl -i "http://$WEB_HOST/"
```

You should see `HTTP/1.1 200 OK` and the HTML page content.

### (Optional) Destroy
```bash
cd ../terraform
terraform destroy
```

If you tell me your distro image family (Ubuntu vs Rocky, etc.) and whether your OpenStack requires a specific way to obtain a floating IP (pool vs network), I can tailor the Terraform/Ansible to match your cloud exactly.
