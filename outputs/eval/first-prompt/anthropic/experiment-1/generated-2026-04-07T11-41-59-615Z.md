### `terraform/main.tf`

```hcl
# -----------------------------------------------
# Terraform Configuration for OpenStack Web Server
# -----------------------------------------------
# This configuration provisions:
#   - A security group allowing SSH (22) and HTTP (80)
#   - A compute instance with the security group applied
#   - A floating IP for external access
# -----------------------------------------------

terraform {
  required_version = ">= 0.14.0"
  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53.0"
    }
  }
}

# -----------------------------------------------
# Provider Configuration
# -----------------------------------------------
# Authentication is handled via environment variables
# (e.g., OS_AUTH_URL, OS_USERNAME, OS_PASSWORD,
#  OS_TENANT_NAME, OS_REGION_NAME) or via clouds.yaml.
# No credentials are hardcoded.
# -----------------------------------------------
provider "openstack" {}

# -----------------------------------------------
# Security Group
# -----------------------------------------------
# Creates a security group with rules permitting
# inbound SSH on port 22 and inbound HTTP on port 80
# from any source (0.0.0.0/0).
# -----------------------------------------------
resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = var.security_group_name
  description = "Security group for web server - allows SSH and HTTP inbound"
}

resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = "Allow inbound SSH access"
}

resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = "Allow inbound HTTP access"
}

# -----------------------------------------------
# Floating IP
# -----------------------------------------------
# Allocates a floating IP from the specified pool
# so the instance is reachable from external networks.
# -----------------------------------------------
resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.floating_ip_pool
}

# -----------------------------------------------
# Compute Instance
# -----------------------------------------------
# Provisions a single virtual machine with the
# specified image, flavor, SSH key, and security group.
# -----------------------------------------------
resource "openstack_compute_instance_v2" "web_server" {
  name            = var.instance_name
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.ssh_key_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  network {
    name = var.network_name
  }
}

# -----------------------------------------------
# Floating IP Association
# -----------------------------------------------
# Associates the allocated floating IP with the
# compute instance so it gets a public address.
# -----------------------------------------------
resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}
```

### `ansible/playbook.yml`

```yaml
---
# -----------------------------------------------
# Ansible Playbook: Configure OpenStack Web Server
# -----------------------------------------------
# This playbook performs the following:
#   1. Installs nginx via the system package manager
#   2. Deploys a static HTML page to /var/www/html/
#   3. Ensures nginx is running as a system service
#      and listening on port 80
# -----------------------------------------------

- name: Configure web server with Nginx
  hosts: webservers
  become: true
  gather_facts: true

  vars_files:
    - vars/main.yml

  tasks:
    # -----------------------------------------
    # Step 1: Install Nginx
    # -----------------------------------------
    # Uses the ansible.builtin.package module which
    # is a generic OS-agnostic package manager
    # abstraction (works with apt, yum, dnf, etc.)
    # The module is idempotent — if nginx is already
    # installed, no action is taken.
    # -----------------------------------------
    - name: Install Nginx web server
      ansible.builtin.package:
        name: "{{ nginx_package_name }}"
        state: present
        update_cache: true

    # -----------------------------------------
    # Step 2: Ensure web root directory exists
    # -----------------------------------------
    - name: Ensure web root directory exists
      ansible.builtin.file:
        path: "{{ web_root }}"
        state: directory
        owner: root
        group: root
        mode: "0755"

    # -----------------------------------------
    # Step 3: Deploy static HTML page
    # -----------------------------------------
    # Renders the Jinja2 template and places the
    # result at the configured web root. A change
    # triggers an Nginx reload via the handler.
    # -----------------------------------------
    - name: Deploy static HTML index page
      ansible.builtin.template:
        src: templates/index.html.j2
        dest: "{{ web_root }}/{{ index_file }}"
        owner: root
        group: root
        mode: "0644"
      notify: Reload Nginx

    # -----------------------------------------
    # Step 4: Ensure Nginx listens on port 80
    # -----------------------------------------
    # The default Nginx configuration already
    # listens on port 80. This task ensures the
    # default server block is present and has the
    # correct root directive.
    # -----------------------------------------
    - name: Configure Nginx default site
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: "0644"
        content: |
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              root {{ web_root }};
              index {{ index_file }};

              server_name _;

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      notify: Reload Nginx

    # -----------------------------------------
    # Step 5: Enable and start Nginx service
    # -----------------------------------------
    # Ensures Nginx is enabled (starts on boot)
    # and is currently running.
    # -----------------------------------------
    - name: Ensure Nginx service is enabled and running
      ansible.builtin.service:
        name: "{{ nginx_service_name }}"
        state: started
        enabled: true

  # -------------------------------------------
  # Handlers
  # -------------------------------------------
  # Handlers run only when notified by a task
  # that reports a change. This avoids unnecessary
  # service restarts on idempotent re-runs.
  # -------------------------------------------
  handlers:
    - name: Reload Nginx
      ansible.builtin.service:
        name: "{{ nginx_service_name }}"
        state: reloaded
```

## Deployment Instructions

### Prerequisites

1. **Terraform** >= 0.14 installed
2. **Ansible** >= 2.14 installed with Python 3.8+
3. **OpenStack credentials** configured via environment variables or `clouds.yaml`
4. An **SSH keypair** already registered in OpenStack (the public key uploaded, the private key stored locally)

### Step 1: Source OpenStack Credentials

Set your OpenStack authentication environment variables. **Do not hardcode credentials in any file.**

```bash
# Option A: Source an OpenStack RC file downloaded from Horizon
source ~/openrc.sh

# Option B: Set environment variables manually
export OS_AUTH_URL="https://your-openstack-endpoint:5000/v3"
export OS_PROJECT_NAME="your-project"
export OS_USERNAME="your-username"
export OS_PASSWORD="your-password"
export OS_USER_DOMAIN_NAME="Default"
export OS_PROJECT_DOMAIN_NAME="Default"
export OS_REGION_NAME="RegionOne"
```

### Step 2: Configure Terraform Variables

```bash
cd terraform/

# Create your tfvars file from the example
cp terraform.tfvars.example terraform.tfvars

# Edit with your values
# At minimum, set ssh_key_name and ssh_private_key_path
vi terraform.tfvars
```

Example `terraform.tfvars` content:
```hcl
ssh_key_name         = "my-keypair"
ssh_private_key_path = "~/.ssh/id_rsa"
image_name           = "Ubuntu-22.04"
flavor_name          = "m1.small"
network_name         = "my-network"
floating_ip_pool     = "external"
ssh_user             = "ubuntu"
```

### Step 3: Provision Infrastructure with Terraform

```bash
# Initialize Terraform — downloads the OpenStack provider
terraform init

# Preview what will be created
terraform plan

# Apply the configuration to create resources
terraform apply
```

When prompted, type `yes` to confirm. Terraform will output the instance IP and connection details.

### Step 4: Generate the Ansible Inventory

After Terraform completes, generate the inventory file from its outputs:

```bash
# Extract Terraform outputs into shell variables
FLOATING_IP=$(terraform output -raw instance_floating_ip)
SSH_USER=$(terraform output -raw ssh_user)
SSH_KEY=$(terraform output -raw ssh_private_key_path)

# Generate the Ansible inventory file
cat > ../ansible/inventory.ini <<EOF
[webservers]
web-server ansible_host=${FLOATING_IP} ansible_user=${SSH_USER} ansible_ssh_private_key_file=${SSH_KEY}

[webservers:vars]
ansible_ssh_common_args='-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null'
EOF

echo "Inventory generated. Target: ${SSH_USER}@${FLOATING_IP}"
```

### Step 5: Wait for the Instance to Be Ready

```bash
# Wait until SSH is reachable (may take 30-60 seconds)
echo "Waiting for SSH to become available on ${FLOATING_IP}..."
until ssh -i ${SSH_KEY} -o StrictHostKeyChecking=no -o ConnectTimeout=5 ${SSH_USER}@${FLOATING_IP} 'echo ready' 2>/dev/null; do
    sleep 5
done
echo "Instance is ready."
```

### Step 6: Configure the Web Server with Ansible

```bash
cd ../ansible/

# Run the playbook
ansible-playbook -i inventory.ini playbook.yml
```

Expected output summary:
```
PLAY RECAP *****************************************************
web-server  : ok=6    changed=4    unreachable=0    failed=0    skipped=0
```

### Step 7: Verify HTTP Access

```bash
# Get the floating IP from Terraform output
cd ../terraform/
FLOATING_IP=$(terraform output -raw instance_floating_ip)

# Test with curl
curl http://${FLOATING_IP}

# Or open in a browser
echo "Open in browser: http://${FLOATING_IP}"
```

You should see the styled HTML page with the heading **"Hello from OpenStack!"** and system metadata (hostname, IP, OS).

### Verify via SSH (Optional)

```bash
# SSH into the instance
$(terraform output -raw ssh_connection_string)

# On the instance, verify Nginx is running and listening on port 80
sudo systemctl status nginx
sudo ss -tlnp | grep ':80'
curl localhost
```
