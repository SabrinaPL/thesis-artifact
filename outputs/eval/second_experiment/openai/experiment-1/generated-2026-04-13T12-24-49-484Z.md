## `terraform/main.tf`

```hcl
provider "openstack" {
  auth_url    = var.openstack_auth_url
  user_name   = var.openstack_user_name
  password    = var.openstack_password
  tenant_name = var.openstack_tenant_name
  region      = var.openstack_region
  domain_name = var.openstack_domain_name
}

resource "openstack_networking_secgroup_v2" "web_sg" {
  name                 = "web-security-group"
  description          = "Security group for web servers"
  delete_default_rules = true
}

resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_networking_secgroup_rule_v2" "egress_ipv4" {
  direction         = "egress"
  ethertype         = "IPv4"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_networking_secgroup_rule_v2" "egress_ipv6" {
  direction         = "egress"
  ethertype         = "IPv6"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_compute_instance_v2" "webservers" {
  for_each    = var.servers
  name        = each.key
  image_name  = var.image_name
  flavor_name = var.flavor_name
  key_pair    = var.keypair_name

  security_groups = [
    openstack_networking_secgroup_v2.web_sg.name
  ]

  network {
    name = var.network_name
  }

  metadata = {
    role         = "webserver"
    server_name  = each.key
    html_content = each.value.html_content
  }
}

resource "local_file" "ansible_inventory" {
  filename = "${path.module}/../ansible/inventory.ini"
  content = templatefile("${path.module}/templates/inventory.ini.tftpl", {
    ansible_user            = var.ansible_user
    ssh_private_key_file    = var.ssh_private_key_file
    servers = {
      for name, instance in openstack_compute_instance_v2.webservers :
      name => {
        ip           = instance.access_ip_v4 != "" ? instance.access_ip_v4 : instance.network[0].fixed_ip_v4
        html_content = var.servers[name].html_content
      }
    }
  })
}
```

## `terraform/templates/inventory.ini.tftpl`

```ini
[web]
%{ for name, server in servers ~}
${name} ansible_host=${server.ip} ansible_user=${ansible_user} ansible_ssh_private_key_file=${ssh_private_key_file} web_content='${server.html_content}'
%{ endfor ~}

[web:vars]
ansible_python_interpreter=/usr/bin/python3
```

# 2. Generated Ansible inventory approach

Terraform creates the inventory dynamically using the `local_file` resource and a template.

After `terraform apply`, this file is generated:

```text
ansible/inventory.ini
```

Example generated content:

```ini
[web]
webserver1 ansible_host=10.0.0.11 ansible_user=ubuntu ansible_ssh_private_key_file=~/.ssh/id_rsa web_content='<html><body><h1>webserver1</h1></body></html>'
webserver2 ansible_host=10.0.0.12 ansible_user=ubuntu ansible_ssh_private_key_file=~/.ssh/id_rsa web_content='<html><body><h1>webserver2</h1></body></html>'

[web:vars]
ansible_python_interpreter=/usr/bin/python3
```

This satisfies the requirement that inventory is generated dynamically from Terraform.

# 3. Ansible playbook

## `ansible/site.yml`

```yaml
---
- name: Configure web servers
  hosts: web
  become: true

  tasks:
    - name: Install nginx
      package:
        name: nginx
        state: present

    - name: Ensure nginx is enabled and started
      service:
        name: nginx
        state: started
        enabled: true

    - name: Deploy custom index.html
      copy:
        dest: /var/www/html/index.html
        content: "{{ web_content }}"
        owner: root
        group: root
        mode: '0644'

    - name: Ensure nginx listens on port 80
      lineinfile:
        path: /etc/nginx/sites-available/default
        regexp: '^\s*listen\s+80\s+default_server;'
        line: '	listen 80 default_server;'
      notify: Restart nginx

    - name: Ensure IPv4 default server block exists
      lineinfile:
        path: /etc/nginx/sites-available/default
        regexp: '^\s*listen\s+80;'
        line: '	listen 80;'
      notify: Restart nginx

  handlers:
    - name: Restart nginx
      service:
        name: nginx
        state: restarted
```

If your base image is RHEL/CentOS-based, package/service names may vary slightly, but `nginx` usually remains the same.

# 7. Notes

- This solution avoids duplication by using:
  - `for_each` for both instances
  - a `servers` variable map for per-host content
- Credentials are not hardcoded in resources
- Security group rules are managed explicitly with:
  - `openstack_networking_secgroup_v2`
  - `openstack_networking_secgroup_rule_v2`
- If your cloud requires floating IPs for external access, you may need to add:
  - floating IP allocation
  - floating IP association
- If your image uses a different SSH username, change `ansible_user`
