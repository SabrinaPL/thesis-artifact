### `terraform/main.tf`

```hcl
terraform {
  required_version = ">= 1.3.0"
  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53.0"
    }
    local = {
      source  = "hashicorp/local"
      version = "~> 2.4"
    }
  }
}

# ---------------------------------------------------------------------------
# Provider
# ---------------------------------------------------------------------------
provider "openstack" {
  user_name   = var.openstack_user_name
  tenant_name = var.openstack_tenant_name
  password    = var.openstack_password
  auth_url    = var.openstack_auth_url
  region      = var.openstack_region
}

# ---------------------------------------------------------------------------
# Security Group & Rules (single group shared by both servers)
# ---------------------------------------------------------------------------
resource "openstack_networking_secgroup_v2" "web_sg" {
  name                 = "web-security-group"
  description          = "Allow inbound SSH and HTTP traffic"
  delete_default_rules = true
}

# Default egress – IPv4
resource "openstack_networking_secgroup_rule_v2" "egress_v4" {
  direction         = "egress"
  ethertype         = "IPv4"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# Default egress – IPv6
resource "openstack_networking_secgroup_rule_v2" "egress_v6" {
  direction         = "egress"
  ethertype         = "IPv6"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# Ingress rules defined as a local map to avoid duplication
locals {
  ingress_rules = {
    ssh = {
      protocol       = "tcp"
      port_range_min = 22
      port_range_max = 22
      description    = "Allow SSH"
    }
    http = {
      protocol       = "tcp"
      port_range_min = 80
      port_range_max = 80
      description    = "Allow HTTP"
    }
  }
}

resource "openstack_networking_secgroup_rule_v2" "ingress" {
  for_each = local.ingress_rules

  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = each.value.protocol
  port_range_min    = each.value.port_range_min
  port_range_max    = each.value.port_range_max
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = each.value.description
}

# ---------------------------------------------------------------------------
# Floating IPs (one per server)
# ---------------------------------------------------------------------------
resource "openstack_networking_floatingip_v2" "fip" {
  for_each = var.webservers
  pool     = var.floating_ip_pool
}

# ---------------------------------------------------------------------------
# Compute Instances
# ---------------------------------------------------------------------------
resource "openstack_compute_instance_v2" "webserver" {
  for_each = var.webservers

  name            = each.value.name
  image_id        = var.image_id
  flavor_name     = var.flavor_name
  key_pair        = var.keypair_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  network {
    uuid = var.network_id
  }

  metadata = {
    role = "webserver"
  }
}

# ---------------------------------------------------------------------------
# Associate Floating IPs
# ---------------------------------------------------------------------------
resource "openstack_compute_floatingip_associate_v2" "fip_assoc" {
  for_each    = var.webservers
  floating_ip = openstack_networking_floatingip_v2.fip[each.key].address
  instance_id = openstack_compute_instance_v2.webserver[each.key].id
}

# ---------------------------------------------------------------------------
# Generate Ansible Inventory
# ---------------------------------------------------------------------------
resource "local_file" "ansible_inventory" {
  filename        = "${path.module}/../ansible/inventory.ini"
  file_permission = "0644"
  content = templatefile("${path.module}/templates/inventory.tpl", {
    webservers = {
      for key, ws in var.webservers : key => {
        ansible_host = openstack_networking_floatingip_v2.fip[key].address
        hostname     = ws.name
      }
    }
    ssh_user = var.ssh_user
  })
}
```

${ws.hostname} ansible_host=${ws.ansible_host} ansible_user=${ssh_user} server_name=${ws.hostname}
%{ endfor ~}

```

### `ansible/playbook.yml`

```yaml
---
- name: Configure web servers with nginx
  hosts: webservers
  become: true
  gather_facts: true

  tasks:

    # ------------------------------------------------------------------
    # 1. Install nginx
    # ------------------------------------------------------------------
    - name: Update apt cache (Debian/Ubuntu)
      ansible.builtin.apt:
        update_cache: yes
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"

    - name: Install nginx (Debian/Ubuntu)
      ansible.builtin.apt:
        name: nginx
        state: present
      when: ansible_os_family == "Debian"

    - name: Install nginx (RedHat/CentOS)
      ansible.builtin.yum:
        name: nginx
        state: present
      when: ansible_os_family == "RedHat"

    # ------------------------------------------------------------------
    # 2. Configure nginx – ensure it listens on port 80 (default)
    # ------------------------------------------------------------------
    - name: Ensure default nginx site is enabled (Debian/Ubuntu)
      ansible.builtin.file:
        src: /etc/nginx/sites-available/default
        dest: /etc/nginx/sites-enabled/default
        state: link
      when: ansible_os_family == "Debian"
      notify: Restart nginx

    # ------------------------------------------------------------------
    # 3. Deploy server-specific HTML content
    # ------------------------------------------------------------------
    - name: Deploy custom index.html
      ansible.builtin.template:
        src: templates/index.html.j2
        dest: /var/www/html/index.html
        owner: root
        group: root
        mode: "0644"
      notify: Restart nginx

    # ------------------------------------------------------------------
    # 4. Ensure nginx is running and enabled
    # ------------------------------------------------------------------
    - name: Ensure nginx is started and enabled
      ansible.builtin.service:
        name: nginx
        state: started
        enabled: true

  handlers:
    - name: Restart nginx
      ansible.builtin.service:
        name: nginx
        state: restarted
```

## Deployment Instructions

### Prerequisites

1. **Terraform** >= 1.3 installed
2. **Ansible** >= 2.12 installed
3. An OpenStack cloud with:
   - A valid image (e.g., Ubuntu 22.04)
   - A network accessible by the instances
   - An SSH keypair registered in OpenStack (private key available locally)
   - A floating IP pool for external access

### Step 1 — Configure Variables

```bash
cd terraform
cp terraform.tfvars.example terraform.tfvars
# Edit terraform.tfvars with your real values
```

> **Important:** Never commit `terraform.tfvars` to version control. Add it to `.gitignore`.

### Step 2 — Provision Infrastructure

```bash
cd terraform

terraform init
terraform validate
terraform plan -out=tfplan
terraform apply tfplan
```

After a successful apply you will see output similar to:

```
webserver_ips = {
  "webserver1" = "203.0.113.10"
  "webserver2" = "203.0.113.11"
}
inventory_file = "../ansible/inventory.ini"
```

The dynamic inventory file is automatically written to `ansible/inventory.ini`.

### Step 3 — Verify Generated Inventory

```bash
cat ../ansible/inventory.ini
```

Expected content:

```ini
[webservers]
webserver1 ansible_host=203.0.113.10 ansible_user=ubuntu server_name=webserver1
webserver2 ansible_host=203.0.113.11 ansible_user=ubuntu server_name=webserver2

[webservers:vars]
ansible_ssh_common_args='-o StrictHostKeyChecking=no'
```

### Step 4 — Configure Servers with Ansible

```bash
cd ../ansible

# Verify connectivity first
ansible -i inventory.ini webservers -m ping --private-key=~/.ssh/your_private_key

# Run the playbook
ansible-playbook -i inventory.ini playbook.yml --private-key=~/.ssh/your_private_key
```

### Step 5 — Verify Each Server Independently

```bash
# Get IPs from Terraform output
cd ../terraform
WS1_IP=$(terraform output -json webserver_ips | jq -r '.webserver1')
WS2_IP=$(terraform output -json webserver_ips | jq -r '.webserver2')

# Test webserver1
curl http://$WS1_IP
# Expected: "Hello from webserver1!"

# Test webserver2
curl http://$WS2_IP
# Expected: "Hello from webserver2!"

# Verify SSH access
ssh -i ~/.ssh/your_private_key ubuntu@$WS1_IP "hostname"
ssh -i ~/.ssh/your_private_key ubuntu@$WS2_IP "hostname"
```

### Step 6 — Tear Down

```bash
cd terraform
terraform destroy
```

## Design Decisions & Notes

| Decision | Rationale |
|---|---|
| `for_each` on `var.webservers` map | Eliminates duplication; adding a third server requires only a map entry |
| `delete_default_rules = true` | Gives Terraform full control over all security group rules |
| Ingress rules via `local.ingress_rules` map | DRY approach — new ports require only a map entry |
| `local_file` for inventory | Inventory is regenerated on every `apply`, always in sync with infrastructure |
| `server_name` host variable in inventory | Passed directly to the Jinja2 template so each server renders unique content |
| Credentials only in `terraform.tfvars` | Never hardcoded; the file is excluded from VCS |
| OS-family conditional in Ansible | Playbook works on both Debian/Ubuntu and RedHat/CentOS images |
```
