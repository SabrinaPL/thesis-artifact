```hcl
# ============================================================
# main.tf — Core Infrastructure for an OpenStack Web Server
# ============================================================
# WHAT THIS FILE DOES:
# 1. Creates a SECURITY GROUP with firewall rules (SSH + HTTP)
# 2. Allocates a FLOATING IP so the server is reachable from
#    the internet
# 3. Provisions a COMPUTE INSTANCE (virtual machine)
# 4. Associates the floating IP with the instance
# 5. Generates an ANSIBLE INVENTORY file so Ansible knows
#    how to connect to the new server
#
# KEY CONCEPT — DECLARATIVE vs IMPERATIVE:
# Terraform is DECLARATIVE: you describe the DESIRED STATE
# ("I want a VM with these properties") and Terraform figures
# out what API calls to make. This is different from imperative
# scripts ("step 1: create network, step 2: create VM...").
# ============================================================

# ------------------------------------------------------------
# Terraform Settings Block
# ------------------------------------------------------------
# This block pins the Terraform CLI version and declares which
# "providers" (plugins that talk to cloud APIs) we need.
# BEST PRACTICE: Always pin versions to avoid breaking changes
# when a new provider version is released.
# ------------------------------------------------------------
terraform {
  required_version = ">= 1.3.0"

  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.54"
      # The "~>" operator means "any version >= 1.54.0 and < 2.0.0".
      # This allows patch/minor updates but prevents major breaking changes.
    }
  }

  # ---------------------------------------------------------------
  # STATE BACKEND (commented out — uncomment for team/production use)
  # ---------------------------------------------------------------
  # WHAT IS STATE?
  # Terraform stores a "state file" that maps your .tf configuration
  # to real cloud resources. By default this is a local file called
  # terraform.tfstate. This is fine for learning, but in a team
  # environment you MUST use a remote backend so everyone shares
  # the same state and can lock it to prevent concurrent changes.
  #
  # Example using an S3-compatible backend (like OpenStack Swift):
  # backend "s3" {
  #   bucket         = "terraform-state"
  #   key            = "web-server/terraform.tfstate"
  #   region         = "RegionOne"
  #   encrypt        = true          # Encrypt state at rest
  #   dynamodb_table = "tf-lock"     # Enables state locking
  # }
  # ---------------------------------------------------------------
}

# ------------------------------------------------------------
# Provider Configuration
# ------------------------------------------------------------
# HOW AUTHENTICATION WORKS:
# We intentionally do NOT put credentials here. Instead, the
# OpenStack provider reads them from environment variables:
#   OS_AUTH_URL, OS_PROJECT_NAME, OS_USERNAME, OS_PASSWORD, etc.
#
# BEST PRACTICE: Never hardcode credentials in Terraform files.
# Use environment variables, a secrets manager, or your CI/CD
# system's secret injection mechanism.
#
# To set these, source your OpenStack RC file:
#   source ~/my-project-openrc.sh
# ------------------------------------------------------------
provider "openstack" {
  region = var.region
}

# ============================================================
# SECURITY GROUP
# ============================================================
# WHAT IS A SECURITY GROUP?
# A security group is a virtual firewall that controls inbound
# and outbound network traffic to your instance. Think of it
# as a bouncer at a club — it decides who gets in and who
# gets out based on rules you define.
#
# PRINCIPLE OF LEAST PRIVILEGE:
# Only open the ports you actually need. Every open port is a
# potential attack surface.
# ============================================================

resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = var.security_group_name
  description = "Security group for web server: allows SSH and HTTP inbound"

  # delete_default_rules = true removes OpenStack's default "allow all
  # egress" rules so we can explicitly define every rule ourselves.
  # This follows the security principle of "deny by default, allow
  # by exception" — nothing is allowed unless we explicitly say so.
  delete_default_rules = true

  # TAGGING: We merge our standard tags with resource-specific ones.
  # Tags don't affect functionality but are crucial for operations:
  # - Finding resources in a large environment
  # - Tracking costs by project/team
  # - Automated compliance scanning
  tags = merge(var.tags, {
    Name        = var.security_group_name
    environment = var.environment
  })
}

# --- Egress (outbound) rules ---
# We allow all outbound traffic so the instance can:
# - Download packages (apt/yum)
# - Reach external APIs
# - Perform DNS lookups
# In high-security environments, you'd restrict this too.

resource "openstack_networking_secgroup_rule_v2" "egress_ipv4" {
  direction         = "egress"
  ethertype         = "IPv4"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

resource "openstack_networking_secgroup_rule_v2" "egress_ipv6" {
  direction         = "egress"
  ethertype         = "IPv6"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# --- Ingress (inbound) rules ---

# SSH ACCESS (Port 22)
# SECURITY NOTE: We use var.allowed_ssh_cidr instead of hardcoding
# "0.0.0.0/0". In production, restrict this to your office IP,
# VPN range, or bastion host CIDR. Leaving SSH open to the entire
# internet is one of the most common security mistakes in cloud
# infrastructure.
resource "openstack_networking_secgroup_rule_v2" "ingress_ssh" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = var.allowed_ssh_cidr # <-- RESTRICTED, not 0.0.0.0/0
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# HTTP ACCESS (Port 80)
# This is intentionally open to the world (0.0.0.0/0) because
# this is a PUBLIC web server — anyone should be able to view it.
resource "openstack_networking_secgroup_rule_v2" "ingress_http" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# ============================================================
# FLOATING IP
# ============================================================
# WHAT IS A FLOATING IP?
# In OpenStack, instances on a "private" network get a private
# IP (like 192.168.x.x) that's not reachable from the internet.
# A floating IP is a public IP address that you "attach" to an
# instance so it becomes externally accessible — similar to an
# Elastic IP in AWS.
# ============================================================

resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.floating_ip_pool

  tags = merge(var.tags, {
    Name        = "${var.instance_name}-fip"
    environment = var.environment
  })
}

# ============================================================
# COMPUTE INSTANCE (Virtual Machine)
# ============================================================
# This is the actual server that will run nginx. Terraform
# creates it by calling the OpenStack Nova API.
#
# IMPORTANT CONCEPT — IMMUTABLE vs MUTABLE INFRASTRUCTURE:
# Terraform provisions the VM (immutable step — create/destroy).
# Ansible then configures it (mutable step — install software,
# copy files). This separation of concerns is a common pattern.
# ============================================================

resource "openstack_compute_instance_v2" "web_server" {
  name              = var.instance_name
  image_name        = var.image_name
  flavor_name       = var.flavor_name
  key_pair          = var.ssh_key_name
  availability_zone = var.availability_zone

  # Attach the security group we created above.
  # We use .name (not .id) because the OpenStack compute API
  # expects the security group name for this particular resource.
  security_groups = [
    openstack_networking_secgroup_v2.web_sg.name
  ]

  network {
    name = var.network_name
  }

  # METADATA: OpenStack's equivalent of tags for compute instances.
  # These key-value pairs are stored with the instance and can be
  # queried via the API or from inside the instance via the
  # metadata service (http://169.254.169.254).
  metadata = merge(var.tags, {
    Name        = var.instance_name
    environment = var.environment
    role        = "web-server"
  })

  # LIFECYCLE: Terraform meta-argument that controls behavior.
  # create_before_destroy ensures that if this instance needs to
  # be replaced (e.g., you change the image), Terraform creates
  # the new one BEFORE destroying the old one, reducing downtime.
  lifecycle {
    create_before_destroy = true
  }
}

# ============================================================
# ASSOCIATE FLOATING IP WITH INSTANCE
# ============================================================
# This is a separate resource because in OpenStack, a floating
# IP and an instance are independent objects that get linked.
# This separation lets you, for example, move a floating IP
# from one instance to another without deallocating it.
# ============================================================

resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}

# ============================================================
# GENERATE ANSIBLE INVENTORY
# ============================================================
# WHY AUTO-GENERATE?
# Terraform knows the IP address of the newly created server.
# By generating the Ansible inventory file automatically, we
# create a seamless handoff between Terraform (infrastructure)
# and Ansible (configuration). No manual copy-pasting of IPs!
#
# The templatefile() function reads a template and substitutes
# variables, similar to Jinja2 templates in Python.
# ============================================================

resource "local_file" "ansible_inventory" {
  content = templatefile("${path.module}/../ansible/inventory.tpl", {
    server_ip        = openstack_networking_floatingip_v2.web_fip.address
    ssh_user         = var.ssh_user
    private_key_path = var.private_key_path
  })
  filename = "${path.module}/../ansible/inventory.ini"

  # File permissions: readable/writable by owner only (600).
  # This is important because the file contains SSH connection
  # details that shouldn't be world-readable.
  file_permission = "0600"

  # depends_on tells Terraform the order of operations:
  # "Don't create the inventory until the floating IP is associated."
  # Without this, the inventory could be generated before the IP
  # is actually usable.
  depends_on = [openstack_compute_floatingip_associate_v2.web_fip_assoc]
}

# ============================================================
# Ansible Inventory (auto-generated by Terraform)
# ============================================================
# WHAT IS AN INVENTORY?
# An inventory tells Ansible WHICH servers to manage and HOW
# to connect to them. Think of it as an address book.
#
# DO NOT EDIT THIS FILE MANUALLY — it will be overwritten
# the next time you run 'terraform apply'.
# ============================================================

[webservers]
web-server ansible_host=${server_ip} ansible_user=${ssh_user} ansible_ssh_private_key_file=${private_key_path}

[webservers:vars]
# StrictHostKeyChecking=no: Skips the "are you sure you want to connect?"
# prompt for first-time SSH connections. This is acceptable for
# ephemeral/automated infrastructure, but in high-security environments
# you should pre-populate known_hosts or use SSH certificates instead.
#
# UserKnownHostsFile=/dev/null: Don't save host keys to the known_hosts
# file. Since cloud VMs are frequently created and destroyed, their IP
# addresses get reused and old host keys would cause connection errors.
ansible_ssh_common_args='-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o ConnectTimeout=30'

# ============================================================
# playbook.yml — Configure a Web Server with Nginx
# ============================================================
# WHAT IS A PLAYBOOK?
# An Ansible playbook is a YAML file that describes a series of
# tasks to execute on one or more servers. Think of it as a
# recipe: "on the webservers group, do these steps in order."
#
# IDEMPOTENCY:
# Every task in this playbook is IDEMPOTENT — meaning you can
# run it 10 times and the result is the same as running it once.
# Ansible checks the current state and only makes changes when
# needed. This is a core principle of configuration management.
# ============================================================

- name: Configure web server with nginx
  hosts: webservers    # <-- Target the [webservers] group from inventory
  become: true         # <-- Run tasks with sudo (root privileges)
  gather_facts: true   # <-- Collect system info (OS type, IP, etc.)
                       #     We need this for the conditional logic below

  # ---------------------------------------------------------------
  # VARIABLES
  # ---------------------------------------------------------------
  # These variables control the nginx configuration. Defining them
  # here (rather than hardcoding in tasks) means you can override
  # them at runtime: ansible-playbook playbook.yml -e "nginx_listen_port=8080"
  # ---------------------------------------------------------------
  vars:
    nginx_document_root: /var/www/html
    nginx_index_file: index.html
    nginx_listen_port: 80
    nginx_server_tokens: "off"  # NEW: Hide nginx version in HTTP headers
                                # (security hardening — don't reveal software versions)

  # ---------------------------------------------------------------
  # PRE-TASKS: Run before main tasks
  # ---------------------------------------------------------------
  # We use pre_tasks for the SSH connectivity check because it
  # must succeed before anything else makes sense.
  # ---------------------------------------------------------------
  pre_tasks:
    # Wait for SSH to become available. After Terraform creates
    # the VM, it takes time for the OS to boot and start sshd.
    # Without this, subsequent tasks would fail with connection errors.
    - name: Wait for SSH to become available
      ansible.builtin.wait_for_connection:
        delay: 10        # Wait 10 seconds before first attempt
        timeout: 300     # Give up after 5 minutes

  tasks:
    # ==============================================================
    # PACKAGE INSTALLATION
    # ==============================================================
    # The 'when' conditionals make this playbook work on BOTH
    # Debian/Ubuntu (apt) and RedHat/CentOS (yum) systems.
    # The ansible_os_family fact is gathered automatically.
    # ==============================================================

    - name: Update apt cache (Debian/Ubuntu)
      ansible.builtin.apt:
        update_cache: yes
        cache_valid_time: 3600   # Don't re-update if cache is < 1 hour old
      when: ansible_os_family == "Debian"

    - name: Install nginx (Debian/Ubuntu)
      ansible.builtin.apt:
        name: nginx
        state: present           # "present" = install if not already installed
                                 # Use "latest" to also upgrade existing installs
      when: ansible_os_family == "Debian"

    - name: Install EPEL repository (RedHat/CentOS)
      ansible.builtin.yum:
        name: epel-release       # nginx is in the EPEL repo for RHEL systems
        state: present
      when: ansible_os_family == "RedHat"

    - name: Install nginx (RedHat/CentOS)
      ansible.builtin.yum:
        name: nginx
        state: present
      when: ansible_os_family == "RedHat"

    # ==============================================================
    # ENSURE DOCUMENT ROOT EXISTS
    # ==============================================================
    # The 'file' module manages files and directories. 'state: directory'
    # means "create this directory if it doesn't exist."
    # We set explicit ownership and permissions as a security practice.
    # ==============================================================

    - name: Create document root directory
      ansible.builtin.file:
        path: "{{ nginx_document_root }}"
        state: directory
        owner: root
        group: root
        mode: "0755"            # Owner: rwx, Group: r-x, Other: r-x

    # ==============================================================
    # REMOVE DEFAULT NGINX PAGE
    # ==============================================================
    # Some nginx packages ship with a default welcome page. We remove
    # it to avoid confusion and ensure only our content is served.
    # ==============================================================

    - name: Remove default nginx index page if different from ours
      ansible.builtin.file:
        path: "{{ nginx_document_root }}/index.nginx-debian.html"
        state: absent
      when: ansible_os_family == "Debian"

    # ==============================================================
    # DEPLOY STATIC HTML PAGE
    # ==============================================================
    # The 'copy' module copies a file from the Ansible control machine
    # to the remote server. The 'notify' keyword triggers the handler
    # only when the file ACTUALLY CHANGES (idempotency in action!).
    # ==============================================================

    - name: Deploy static index.html
      ansible.builtin.copy:
        src: files/index.html
        dest: "{{ nginx_document_root }}/{{ nginx_index_file }}"
        owner: root
        group: root
        mode: "0644"            # Owner: rw-, Group: r--, Other: r--
        # SECURITY: Web-served files should be readable but NOT
        # writable by the web server process. nginx runs as www-data
        # (which is in "Other"), so r-- is sufficient.
      notify: Reload nginx

    # ==============================================================
    # NGINX CONFIGURATION
    # ==============================================================
    # We deploy a custom nginx configuration that:
    # 1. Listens on the specified port (default: 80)
    # 2. Serves files from our document root
    # 3. Disables server_tokens (hides nginx version — security best practice)
    # ==============================================================

    - name: Deploy nginx default site configuration (Debian/Ubuntu)
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: "0644"
        content: |
          # Managed by Ansible — DO NOT EDIT MANUALLY
          # Manual changes will be overwritten on next Ansible run.
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              root {{ nginx_document_root }};
              index {{ nginx_index_file }};

              # '_' matches any hostname (catch-all server block)
              server_name _;

              # Hide nginx version number in error pages and headers.
              # Revealing software versions helps attackers find known
              # vulnerabilities for that specific version.
              server_tokens {{ nginx_server_tokens }};

              # Prevent clickjacking attacks by disallowing this site
              # from being embedded in iframes on other domains.
              add_header X-Frame-Options "SAMEORIGIN" always;

              # Prevent MIME-type sniffing which can lead to XSS attacks.
              add_header X-Content-Type-Options "nosniff" always;

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      when: ansible_os_family == "Debian"
      notify: Reload nginx

    - name: Deploy nginx configuration (RedHat/CentOS)
      ansible.builtin.copy:
        dest: /etc/nginx/conf.d/default.conf
        owner: root
        group: root
        mode: "0644"
        content: |
          # Managed by Ansible — DO NOT EDIT MANUALLY
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              root {{ nginx_document_root }};
              index {{ nginx_index_file }};

              server_name _;
              server_tokens {{ nginx_server_tokens }};

              add_header X-Frame-Options "SAMEORIGIN" always;
              add_header X-Content-Type-Options "nosniff" always;

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      when: ansible_os_family == "RedHat"
      notify: Reload nginx

    # ==============================================================
    # VALIDATE NGINX CONFIGURATION
    # ==============================================================
    # Before reloading nginx, we validate the configuration file.
    # This prevents a broken config from taking down the web server.
    # 'changed_when: false' tells Ansible this is a read-only check
    # that doesn't modify anything.
    # ==============================================================

    - name: Validate nginx configuration
      ansible.builtin.command: nginx -t
      changed_when: false

    # ==============================================================
    # ENABLE AND START NGINX SERVICE
    # ==============================================================
    # 'enabled: true' means nginx starts automatically after reboot.
    # 'state: started' means start it now if it's not already running.
    # ==============================================================

    - name: Enable and start nginx service
      ansible.builtin.systemd:
        name: nginx
        enabled: true
        state: started

  # ================================================================
  # HANDLERS
  # ================================================================
  # Handlers are tasks that only run when NOTIFIED by another task.
  # They always run at the END of the play (after all tasks), and
  # they run only ONCE even if notified multiple times.
  #
  # WHY USE HANDLERS?
  # If we reloaded nginx after EVERY config change task, we'd
  # restart it multiple times unnecessarily. Handlers consolidate
  # this into a single reload at the end.
  # ================================================================

  handlers:
    - name: Reload nginx
      ansible.builtin.systemd:
        name: nginx
        state: reloaded         # 'reloaded' is gentler than 'restarted' —
                                # it re-reads config without dropping existing
                                # connections.
```

## 3. Deployment Instructions (Updated)

### Prerequisites

| Tool | Minimum Version | Purpose |
|------|-----------------|---------|
| Terraform | >= 1.3 | Provisions cloud infrastructure |
| Ansible | >= 2.12 | Configures the server |
| OpenStack CLI | any (optional) | Useful for debugging and keypair management |

### Step 1: Set OpenStack Credentials

```bash
# Option A: Source your OpenStack RC file (downloaded from Horizon dashboard)
source ~/my-project-openrc.sh

# Option B: Set environment variables manually
export OS_AUTH_URL="https://your-openstack:5000/v3"
export OS_PROJECT_NAME="my-project"
export OS_USER_DOMAIN_NAME="Default"
export OS_PROJECT_DOMAIN_NAME="Default"
export OS_USERNAME="my-username"
export OS_PASSWORD="my-password"   # Consider using a credential helper instead
export OS_REGION_NAME="RegionOne"
export OS_IDENTITY_API_VERSION=3
```

> ⚠️ **SECURITY TIP:** Avoid storing `OS_PASSWORD` in shell history. Use `read -s OS_PASSWORD && export OS_PASSWORD` or a credential helper.

### Step 2: Register Your SSH Key (if not done already)

```bash
# Check existing keypairs
openstack keypair list

# Create one from your existing public key
openstack keypair create --public-key ~/.ssh/id_rsa.pub my-keypair
```

### Step 3: Configure Terraform Variables

```bash
cd terraform/
cp terraform.tfvars.example terraform.tfvars
```

Edit `terraform.tfvars`:

```hcl
ssh_key_name     = "my-keypair"
private_key_path = "~/.ssh/id_rsa"

# IMPORTANT: Restrict SSH to your IP for security
allowed_ssh_cidr = "YOUR.PUBLIC.IP/32"    # Find it: curl ifconfig.me
```

### Step 4: Provision Infrastructure

```bash
cd terraform/

# Download provider plugins
terraform init

# Format and validate (best practice before every apply)
terraform fmt -check       # Check formatting
terraform validate         # Validate syntax and logic

# Preview what will be created
terraform plan

# Create the resources (type 'yes' when prompted)
terraform apply
```

### Step 5: Configure with Ansible

```bash
cd ../ansible/

# Verify inventory was generated
cat inventory.ini

# Run the playbook
ansible-playbook -i inventory.ini playbook.yml
```

Expected output: All tasks show `ok` or `changed`.

### Step 6: Verify

```bash
# Get the floating IP
FLOATING_IP=$(cd ../terraform && terraform output -raw floating_ip)

# Test HTTP
curl -v http://${FLOATING_IP}

# Or open in browser
echo "Visit: http://${FLOATING_IP}"
```

### Teardown

```bash
cd terraform/
terraform destroy
```

## 4. Summary of Improvements

| # | Change | Rationale |
|---|--------|-----------|
| 1 | Added `allowed_ssh_cidr` variable with validation | **Security**: SSH from 0.0.0.0/0 is the #1 misconfiguration in cloud security. Now operators can (and are strongly encouraged to) restrict it. |
| 2 | Added `.gitignore` | **Security/VCS**: Prevents accidental commit of state files (may contain secrets), tfvars (may contain environment-specific values), and generated inventory files. |
| 3 | Added `sensitive = true` to `private_key_path` and `ssh_command` output | **Security**: Prevents Terraform from displaying filesystem paths and connection details in console output and logs. |
| 4 | Added `file_permission = "0600"` on generated inventory | **Security**: The inventory file contains SSH connection details; it should only be readable by the owner. |
| 5 | Added resource tagging (`tags`, `metadata`) | **Governance**: Per best practices, consistent tagging enables cost tracking, ownership identification, and automated compliance scanning. |
| 6 | Added `environment` variable with validation | **Code Quality**: Constrains allowed values and enables environment-aware tagging. |
| 7 | Added validation blocks on critical variables | **Code Quality**: Catches errors at `terraform plan` time rather than during API calls, following the "fail fast" principle. |
| 8 | Added `wait_for_connection` pre-task in Ansible | **Reliability**: The original solution relied on manual waiting. This makes it automated and robust. |
| 9 | Added `nginx -t` validation task | **Reliability**: Validates nginx config before reload, preventing accidental service outage from a typo. |
| 10 | Added security headers (`X-Frame-Options`, `X-Content-Type-Options`, `server_tokens off`) | **Security hardening**: Prevents common web vulnerabilities (clickjacking, MIME sniffing) and hides server version information. |
| 11 | Added EPEL repository task for RedHat | **Reliability**: nginx isn't in default RHEL repos; without EPEL, the install would fail on CentOS/RHEL. |
| 12 | Added removal of default nginx page | **Correctness**: Prevents the default Debian nginx page from being served alongside our custom page. |
| 13 | Added `lifecycle { create_before_destroy = true }` | **Availability**: Reduces downtime during instance replacement by creating the new VM before destroying the old one. |
| 14 | Added `ConnectTimeout=30` to SSH args | **Reliability**: Prevents Ansible from hanging indefinitely on network issues. |
| 15 | Added extensive inline comments throughout | **Education**: Every resource, variable, and concept is explained for junior developers with no prior IaC knowledge. |
| 16 | Added state backend documentation (commented out) | **Best Practice / Education**: Teaches the importance of remote state for team environments while not breaking the single-developer workflow. |
