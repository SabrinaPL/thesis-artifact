# =============================================
# MAIN TERRAFORM CONFIGURATION
# =============================================
# This file is the heart of our infrastructure
# definition. It tells Terraform WHAT to create
# in OpenStack.
#
# Terraform is "declarative" — you describe the
# desired end state, and Terraform figures out
# the steps to get there. This is different from
# "imperative" scripting where you write step-by-
# step instructions.
#
# WHAT THIS CREATES:
#   1. A security group (firewall rules)
#   2. A virtual machine (compute instance)
#   3. A floating IP (public IP address)
#   4. An association between the IP and the VM
# =============================================

# ---------------------------------------------
# TERRAFORM SETTINGS BLOCK
# ---------------------------------------------
# This block configures Terraform itself, not
# your infrastructure. It specifies:
# - The minimum Terraform version required
# - Which "providers" (cloud plugins) to use
#   and their version constraints
#
# BEST PRACTICE: Always pin provider versions.
# Without pinning, Terraform might download a
# newer version that introduces breaking changes.
# The "~>" operator means "allow patch updates
# only" (e.g., 1.53.0 -> 1.53.9, but not 1.54.0).
# ---------------------------------------------
terraform {
  required_version = ">= 1.0.0"

  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.53"
    }
  }

  # NOTE FOR PRODUCTION: You should configure a
  # "backend" to store Terraform state remotely
  # (e.g., in an S3-compatible object store or
  # Consul). Local state files are fine for learning
  # but problematic for teams because:
  # - They can't be shared between team members
  # - They might contain sensitive data
  # - They can be lost if your laptop dies
  #
  # Example (uncomment and configure for production):
  # backend "s3" {
  #   bucket   = "my-terraform-state"
  #   key      = "web-server/terraform.tfstate"
  #   endpoint = "https://my-s3-endpoint"
  # }
}

# ---------------------------------------------
# PROVIDER CONFIGURATION
# ---------------------------------------------
# A "provider" is Terraform's plugin for talking
# to a specific cloud or service. The OpenStack
# provider knows how to create VMs, networks,
# security groups, etc. via the OpenStack API.
#
# SECURITY: We leave this block empty on purpose!
# Authentication is handled via environment
# variables (OS_AUTH_URL, OS_USERNAME, OS_PASSWORD,
# OS_PROJECT_NAME, etc.) or a clouds.yaml file.
#
# NEVER put credentials in Terraform files because:
# 1. Files get committed to version control (Git)
# 2. Anyone with repo access sees your password
# 3. Credentials in code are the #1 security mistake
# ---------------------------------------------
provider "openstack" {
  # All auth comes from environment variables.
  # See deployment instructions for setup.
}

# ---------------------------------------------
# LOCAL VALUES
# ---------------------------------------------
# "locals" let you compute values once and reuse
# them throughout the configuration. They're like
# constants or helper variables. Here we merge
# user-provided tags with auto-generated ones.
# ---------------------------------------------
locals {
  # Merge user-provided tags with some defaults
  # that are always useful for tracking resources.
  common_tags = merge(var.tags, {
    environment = var.environment
    managed_by  = "terraform"
  })
}

# =============================================
# SECURITY GROUP
# =============================================
# A security group is like a virtual firewall
# that controls which network traffic can reach
# your instance. Without one, your VM might be
# unreachable — or dangerously exposed.
#
# CONCEPT: "Principle of Least Privilege"
# Only open the ports you actually need. Each
# open port is a potential attack surface.
# We need exactly two:
#   - Port 22 (SSH) for remote administration
#   - Port 80 (HTTP) for serving web pages
# =============================================

resource "openstack_networking_secgroup_v2" "web_sg" {
  name        = "${var.environment}-${var.security_group_name}"
  description = "Security group for ${var.instance_name}: allows SSH (22) and HTTP (80) inbound"

  # 'tags' help you identify, filter, and track
  # resources across your cloud environment.
  # BEST PRACTICE from retrieved docs:
  # "Tag resources consistently — apply a uniform
  #  tag set for governance, auditing, and cost tracking"
  tags = [for k, v in local.common_tags : "${k}=${v}"]
}

# SECURITY GROUP RULE: Allow SSH
# --------------------------------
# This rule allows incoming TCP connections on
# port 22 (SSH) from the specified CIDR range.
#
# SECURITY WARNING: If allowed_ssh_cidr is set
# to "0.0.0.0/0", ANYONE on the internet can
# attempt to SSH into your server. In production,
# ALWAYS restrict this to your office/VPN IP.
# Example: "203.0.113.50/32" (/32 = single IP)
resource "openstack_networking_secgroup_rule_v2" "ssh_ingress" {
  direction         = "ingress"    # "ingress" = incoming traffic
  ethertype         = "IPv4"       # IPv4 (vs IPv6)
  protocol          = "tcp"        # SSH uses TCP
  port_range_min    = 22           # Start of port range
  port_range_max    = 22           # End of port range (same = single port)
  remote_ip_prefix  = var.allowed_ssh_cidr  # WHO can connect
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = "Allow inbound SSH from ${var.allowed_ssh_cidr}"
}

# SECURITY GROUP RULE: Allow HTTP
# --------------------------------
# This rule allows incoming TCP connections on
# port 80 (HTTP) from anywhere. This is expected
# for a public web server — your website needs
# to be accessible to everyone.
resource "openstack_networking_secgroup_rule_v2" "http_ingress" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"  # Open to everyone (public web server)
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
  description       = "Allow inbound HTTP from anywhere"
}

# =============================================
# FLOATING IP
# =============================================
# In OpenStack, instances on private networks
# get private IPs (like 192.168.x.x) that aren't
# reachable from the internet. A "floating IP"
# is a public IP address that you allocate from
# a pool and attach to an instance, making it
# externally accessible.
#
# Analogy: A floating IP is like a phone number
# forwarding service — traffic to the public IP
# gets forwarded to your private instance.
# =============================================
resource "openstack_networking_floatingip_v2" "web_fip" {
  pool        = var.floating_ip_pool
  description = "Floating IP for ${var.instance_name}"

  # NOTE: In some OpenStack environments, you might
  # need to use 'openstack_networking_floatingip_v2'
  # with a 'subnet_id' instead of just 'pool'.
}

# =============================================
# COMPUTE INSTANCE (Virtual Machine)
# =============================================
# This is the actual server! Terraform tells
# OpenStack to create a VM with the specified:
# - Image (OS template, like Ubuntu 22.04)
# - Flavor (size: CPU count, RAM, disk)
# - Key pair (SSH public key for login)
# - Security group (firewall rules)
# - Network (which virtual network to join)
#
# CONCEPT: "Infrastructure as Code" means this
# VM definition is version-controlled, reviewable,
# repeatable, and destroyable. No clicking around
# in a web console!
# =============================================
resource "openstack_compute_instance_v2" "web_server" {
  name            = "${var.environment}-${var.instance_name}"
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.ssh_key_name
  security_groups = [openstack_networking_secgroup_v2.web_sg.name]

  # Attach the instance to the specified network.
  # The instance will get a private IP from this network.
  network {
    name = var.network_name
  }

  # Metadata is OpenStack's way of tagging instances.
  # It appears in the Nova API and Horizon dashboard.
  metadata = local.common_tags

  # 'depends_on' creates an explicit ordering:
  # Terraform will create security group rules BEFORE
  # the instance. Normally Terraform infers dependencies
  # from resource references, but 'depends_on' is useful
  # when the dependency isn't visible in the code.
  depends_on = [
    openstack_networking_secgroup_rule_v2.ssh_ingress,
    openstack_networking_secgroup_rule_v2.http_ingress,
  ]
}

# =============================================
# FLOATING IP ASSOCIATION
# =============================================
# This connects the floating (public) IP to our
# instance. After this, the instance is reachable
# from the internet at the floating IP address.
# =============================================
resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id

  # The floating IP association depends on both the
  # IP being allocated AND the instance being created.
  # Terraform usually figures this out from the
  # references above, but being explicit doesn't hurt.
}

### `ansible/playbook.yml`

```yaml
---
# =============================================
# ANSIBLE PLAYBOOK: Configure Web Server
# =============================================
# An Ansible "playbook" is a YAML file that
# describes a series of tasks to perform on
# remote servers. It's the Ansible equivalent
# of a Terraform configuration — but while
# Terraform provisions infrastructure (VMs,
# networks), Ansible CONFIGURES it (installs
# software, edits files, manages services).
#
# This playbook:
#   1. Updates the package cache
#   2. Installs Nginx (web server)
#   3. Creates the web root directory
#   4. Deploys a static HTML page from a template
#   5. Configures Nginx to serve the page on port 80
#   6. Ensures Nginx starts on boot
#
# CONCEPT: "Idempotency"
# Every task is written so that running the
# playbook 1 time or 100 times produces the
# same result. If Nginx is already installed,
# the install task does nothing. This is a
# core principle of both Ansible and Terraform.
# =============================================

- name: Configure web server with Nginx
  hosts: webservers
  become: true
  # 'become: true' means "run tasks as root"
  # (like using 'sudo'). Package installation
  # and service management require root privileges.

  gather_facts: true
  # Ansible connects to the host and collects
  # system information ("facts"): OS type, IP
  # addresses, CPU count, memory, etc. We use
  # some of these facts in our HTML template.

  vars_files:
    - vars/main.yml
    # Load our variables from the external file.
    # This keeps the playbook clean and variables
    # easy to find and modify.

  # -----------------------------------------
  # PRE-TASKS
  # -----------------------------------------
  # Tasks that run before the main task list.
  # Good for prerequisites like updating the
  # package cache (equivalent of 'apt update').
  # -----------------------------------------
  pre_tasks:
    - name: Update apt package cache (Debian/Ubuntu)
      ansible.builtin.apt:
        update_cache: true
        cache_valid_time: 3600
        # 'cache_valid_time: 3600' means "don't update
        # the cache if it was updated less than 1 hour ago".
        # This speeds up repeated playbook runs.
      when: ansible_os_family == "Debian"
      # 'when' is a conditional — this task only runs
      # on Debian-based systems (Ubuntu, Debian).
      # On CentOS/RHEL, the package module handles
      # cache updates differently.

  # -----------------------------------------
  # MAIN TASKS
  # -----------------------------------------
  tasks:
    # ----- Task 1: Install Nginx -----
    # 'ansible.builtin.package' is a generic module
    # that works across different Linux distributions.
    # It automatically uses 'apt' on Ubuntu, 'yum'
    # on CentOS 7, 'dnf' on CentOS 8+, etc.
    - name: Install Nginx web server
      ansible.builtin.package:
        name: "{{ nginx_package_name }}"
        state: present
        # 'state: present' = "ensure it's installed"
        # 'state: absent'  = "ensure it's NOT installed"
        # 'state: latest'  = "ensure latest version"
        # We use 'present' (not 'latest') for stability
        # — we don't want surprise upgrades in production.

    # ----- Task 2: Create web root directory -----
    # Ensure the directory where our HTML will live
    # exists with the correct permissions.
    - name: Ensure web root directory exists
      ansible.builtin.file:
        path: "{{ web_root }}"
        state: directory
        owner: root
        group: root
        mode: "0755"
        # Permissions: owner=rwx, group=rx, others=rx
        # Web root should be readable by all (Nginx
        # runs as 'www-data' or 'nginx' user) but
        # only writable by root.

    # ----- Task 3: Deploy HTML page from template -----
    # The 'template' module renders a Jinja2 template
    # and copies the result to the remote server.
    # Unlike 'copy', 'template' processes {{ variables }}.
    - name: Deploy static HTML index page
      ansible.builtin.template:
        src: templates/index.html.j2
        dest: "{{ web_root }}/{{ index_file }}"
        owner: root
        group: root
        mode: "0644"
        # '0644' = owner can read/write, everyone else
        # can only read. Standard for web content files.
      notify: Reload Nginx
      # 'notify' tells Ansible: "if this task changes
      # something, trigger the 'Reload Nginx' handler."
      # If the HTML file is already identical, the task
      # reports 'ok' (no change) and the handler doesn't run.

    # ----- Task 4: Configure Nginx server block -----
    # This replaces the default Nginx configuration
    # with one that serves our HTML page on port 80.
    - name: Configure Nginx default site
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: "0644"
        content: |
          # Managed by Ansible — do not edit manually.
          # Manual changes will be overwritten on next
          # playbook run.
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;
              # 'listen [::]:80' enables IPv6 support.
              # 'default_server' means this is the fallback
              # for requests that don't match any other
              # server block.

              root {{ web_root }};
              index {{ index_file }};

              server_name _;
              # 'server_name _;' is a catch-all — matches
              # any hostname. Fine for a single-site server.

              location / {
                  try_files $uri $uri/ =404;
                  # Serve the requested file, or a directory
                  # index, or return 404 Not Found.
              }
          }
      notify: Reload Nginx

    # ----- Task 5: Ensure sites-enabled symlink exists -----
    # On Debian/Ubuntu, Nginx uses a sites-available/
    # sites-enabled pattern. The config in sites-available
    # is only active if there's a symlink in sites-enabled.
    - name: Ensure default site is enabled (Debian/Ubuntu)
      ansible.builtin.file:
        src: /etc/nginx/sites-available/default
        dest: /etc/nginx/sites-enabled/default
        state: link
        # 'state: link' creates a symbolic link.
      when: ansible_os_family == "Debian"
      notify: Reload Nginx

    # ----- Task 6: Validate Nginx configuration -----
    # BEST PRACTICE: Always validate config before
    # reloading a service. A syntax error in the Nginx
    # config would cause the reload to fail, potentially
    # taking your site down.
    - name: Validate Nginx configuration
      ansible.builtin.command:
        cmd: nginx -t
      changed_when: false
      # 'changed_when: false' tells Ansible this command
      # never "changes" anything — it's just a check.
      # Without this, Ansible would report 'changed' every
      # time, which is misleading and breaks idempotency.

    # ----- Task 7: Start and enable Nginx -----
    # 'enabled: true' = start automatically on boot
    # 'state: started' = ensure it's running right now
    - name: Ensure Nginx service is enabled and running
      ansible.builtin.service:
        name: "{{ nginx_service_name }}"
        state: started
        enabled: true

  # -----------------------------------------
  # HANDLERS
  # -----------------------------------------
  # Handlers are special tasks that only run
  # when "notified" by another task that made
  # a change. They run ONCE at the end of the
  # play, even if notified multiple times.
  #
  # WHY use handlers instead of regular tasks?
  # If you changed both the HTML and the config,
  # you'd want Nginx reloaded ONCE (not twice).
  # Handlers deduplicate automatically.
  # -----------------------------------------
  handlers:
    - name: Reload Nginx
      ansible.builtin.service:
        name: "{{ nginx_service_name }}"
        state: reloaded
        # 'reloaded' re-reads config without dropping
        # existing connections. Safer than 'restarted'
        # which fully stops and starts the service.
```

### `ansible/inventory.ini.tpl`

```ini
# =============================================
# ANSIBLE INVENTORY TEMPLATE
# =============================================
# An "inventory" tells Ansible WHICH servers to
# manage. It maps hostnames to connection details.
#
# This is a TEMPLATE — replace the placeholders
# with actual values from Terraform outputs.
# Better yet, use the automated script in the
# deployment instructions to generate this file.
#
# FORMAT:
# [group_name]
# alias  ansible_host=IP  ansible_user=USER  ansible_ssh_private_key_file=KEY
# =============================================

[webservers]
web-server ansible_host=FLOATING_IP ansible_user=SSH_USER ansible_ssh_private_key_file=SSH_PRIVATE_KEY_PATH

[webservers:vars]
# These variables apply to all hosts in the
# 'webservers' group.

# SECURITY NOTE: 'StrictHostKeyChecking=no' disables
# SSH host key verification. This is convenient for
# automation (new servers have unknown host keys) but
# means you won't detect man-in-the-middle attacks.
# For production, consider pre-populating known_hosts
# or using SSH certificates instead.
ansible_ssh_common_args='-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null'

# Connection timeout (seconds) — fail fast if host
# is unreachable rather than waiting forever.
ansible_ssh_timeout=30

## Improved Deployment Instructions

### Prerequisites

1. **Terraform** >= 1.0 installed ([install guide](https://developer.hashicorp.com/terraform/install))
2. **Ansible** >= 2.14 installed (`pip install ansible`)
3. **OpenStack credentials** configured via environment variables or `clouds.yaml`
4. An **SSH keypair** registered in OpenStack (public key uploaded; private key stored locally)

### Step 1: Source OpenStack Credentials

```bash
# SECURITY: Never store credentials in files that get committed to Git.
# Use one of these methods:

# Option A: Source an OpenStack RC file from Horizon dashboard
source ~/openrc.sh

# Option B: Set environment variables manually
export OS_AUTH_URL="https://your-openstack:5000/v3"
export OS_PROJECT_NAME="your-project"
export OS_USERNAME="your-username"
export OS_PASSWORD="your-password"       # Consider using a prompt instead
export OS_USER_DOMAIN_NAME="Default"
export OS_PROJECT_DOMAIN_NAME="Default"
export OS_REGION_NAME="RegionOne"

# Option C (more secure): Prompt for password
read -sr OS_PASSWORD_INPUT && export OS_PASSWORD="$OS_PASSWORD_INPUT"
```

### Step 2: Configure Terraform Variables

```bash
cd terraform/

# Create your variables file from the example
cp terraform.tfvars.example terraform.tfvars

# Edit with YOUR values — at minimum, set:
#   ssh_key_name, ssh_private_key_path, allowed_ssh_cidr
vi terraform.tfvars
```

### Step 3: Validate and Apply Terraform

```bash
# BEST PRACTICE: Always format and validate before applying.
# This catches syntax errors and inconsistencies BEFORE
# they hit your cloud environment.

# Initialize — downloads the OpenStack provider plugin
terraform init

# Format code consistently (auto-fixes style issues)
terraform fmt -recursive

# Validate syntax and configuration
terraform validate

# Preview what will be created (dry run)
# Always review this output carefully!
terraform plan

# Create the infrastructure
# Type 'yes' when prompted to confirm
terraform apply
```

### Step 4: Generate Ansible Inventory

```bash
# Extract outputs into shell variables
FLOATING_IP=$(terraform output -raw instance_floating_ip)
SSH_USER=$(terraform output -raw ssh_user)
SSH_KEY=$(terraform output -raw ssh_private_key_path)

# Generate inventory file from Terraform outputs
cat > ../ansible/inventory.ini <<EOF
[webservers]
web-server ansible_host=${FLOATING_IP} ansible_user=${SSH_USER} ansible_ssh_private_key_file=${SSH_KEY}

[webservers:vars]
ansible_ssh_common_args='-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null'
ansible_ssh_timeout=30
EOF

echo "✅ Inventory generated: ${SSH_USER}@${FLOATING_IP}"
```

### Step 5: Wait for Instance Readiness

```bash
# Cloud instances take time to boot, initialize cloud-init,
# and start the SSH daemon. This loop waits until SSH responds.
echo "⏳ Waiting for SSH on ${FLOATING_IP}..."
RETRIES=0
MAX_RETRIES=24  # 24 × 5 seconds = 2 minutes max wait
until ssh -i "${SSH_KEY}" \
         -o StrictHostKeyChecking=no \
         -o ConnectTimeout=5 \
         "${SSH_USER}@${FLOATING_IP}" 'echo ready' 2>/dev/null; do
    RETRIES=$((RETRIES + 1))
    if [ $RETRIES -ge $MAX_RETRIES ]; then
        echo "❌ Timeout waiting for SSH. Check instance console in Horizon."
        exit 1
    fi
    sleep 5
done
echo "✅ Instance is ready."
```

### Step 6: Run Ansible Playbook

```bash
cd ../ansible/

# Syntax check first (catches YAML errors without connecting)
ansible-playbook -i inventory.ini playbook.yml --syntax-check

# Dry run (shows what WOULD change without actually changing it)
ansible-playbook -i inventory.ini playbook.yml --check --diff

# Apply the configuration
ansible-playbook -i inventory.ini playbook.yml
```

Expected output:
```
PLAY RECAP *****************************************************
web-server : ok=8    changed=5    unreachable=0    failed=0    skipped=0
```

### Step 7: Verify

```bash
cd ../terraform/
FLOATING_IP=$(terraform output -raw instance_floating_ip)

# Test HTTP access
curl -s http://${FLOATING_IP}

# Or open in a browser
echo "🌐 Open: http://${FLOATING_IP}"

# Verify Nginx is running (optional SSH check)
ssh -i "$(terraform output -raw ssh_private_key_path)" \
    "$(terraform output -raw ssh_user)@${FLOATING_IP}" \
    'sudo systemctl status nginx && sudo ss -tlnp | grep :80'
```

### Teardown

```bash
cd terraform/
terraform destroy
# Type 'yes' to confirm. This removes ALL resources.
```
