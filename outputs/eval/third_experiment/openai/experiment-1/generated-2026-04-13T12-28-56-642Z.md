## `terraform/main.tf`

```hcl
provider "openstack" {
  auth_url    = var.openstack_auth_url
  region      = var.openstack_region
  tenant_name = var.openstack_tenant_name
  user_name   = var.openstack_user_name
  password    = var.openstack_password
}

data "openstack_networking_network_v2" "network" {
  name = var.network_name
}

resource "openstack_compute_secgroup_v2" "web_sg" {
  name        = "web-security-group"
  description = "Security group for internal web servers"

  rule {
    from_port   = 22
    to_port     = 22
    ip_protocol = "tcp"
    cidr        = "0.0.0.0/0"
  }

  rule {
    from_port   = var.web_port
    to_port     = var.web_port
    ip_protocol = "tcp"
    cidr        = var.subnet_cidr
  }
}

resource "openstack_compute_secgroup_v2" "lb_sg" {
  name        = "lb-security-group"
  description = "Security group for nginx load balancer"

  rule {
    from_port   = 22
    to_port     = 22
    ip_protocol = "tcp"
    cidr        = "0.0.0.0/0"
  }

  rule {
    from_port   = var.lb_port
    to_port     = var.lb_port
    ip_protocol = "tcp"
    cidr        = "0.0.0.0/0"
  }
}

resource "openstack_compute_instance_v2" "webserver1" {
  name            = "webserver1"
  image_name      = var.image_name
  flavor_name     = var.flavor_name
  key_pair        = var.keypair_name
  security_groups = [openstack_compute_secgroup_v2.web_sg.name]

  network {
    uuid = data.openstack_networking_network_v2.network.id

resource "openstack_compute_instance_v2" "webserver2" {
```

# 4. Ansible playbook

## `ansible/playbook.yml`

```yaml
---
- name: Configure web servers
  hosts: web
  become: true
  vars:
    server_identity: '{{ inventory_hostname }}'
  tasks:
    - name: Install nginx
      ansible.builtin.package:
        name: '{{ nginx_pkg }}'
        state: present

    - name: Ensure nginx is enabled and started
      ansible.builtin.service:
        name: nginx
        enabled: true
        state: started

    - name: Deploy distinct index page
      ansible.builtin.template:
        src: templates/web-index.html.j2
        dest: /var/www/html/index.html
        owner: root
        group: root
        mode: '0644'
      notify: restart nginx

    - name: Ensure default nginx site listens on configured web port
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        mode: '0644'
        content: |
          server {
              listen {{ web_port }} default_server;
              listen [::]:{{ web_port }} default_server;

              root /var/www/html;
              index index.html index.htm;

              server_name _;

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      notify: restart nginx

  handlers:
    - name: restart nginx
      ansible.builtin.service:
        name: nginx
        state: restarted

- name: Configure load balancer
  hosts: loadbalancer
  become: true
  tasks:
    - name: Install nginx
      ansible.builtin.package:
        name: '{{ nginx_pkg }}'
        state: present

    - name: Remove default nginx site if present
      ansible.builtin.file:
        path: /etc/nginx/sites-enabled/default
        state: absent
      notify: restart nginx

    - name: Deploy nginx load balancer config
      ansible.builtin.template:
        src: templates/nginx-lb.conf.j2
        dest: /etc/nginx/conf.d/loadbalancer.conf
        owner: root
        group: root
        mode: '0644'
      notify: restart nginx

    - name: Ensure nginx is enabled and started
      ansible.builtin.service:
        name: nginx
        enabled: true
        state: started

    - name: Validate nginx configuration
      ansible.builtin.command: nginx -t
      changed_when: false

  handlers:
    - name: restart nginx
      ansible.builtin.service:
        name: nginx
        state: restarted
```

## Load balancer NGINX config: `ansible/templates/nginx-lb.conf.j2`

This uses an `upstream` block with the two web servers’ private IPs and round-robin load balancing by default.

```nginx
upstream backend_app {
    server {{ hostvars['webserver1'].ansible_host }}:{{ web_port }};
    server {{ hostvars['webserver2'].ansible_host }}:{{ web_port }};
}

server {
    listen {{ lb_port }};
    listen [::]:{{ lb_port }};

    server_name _;

    location / {
        proxy_pass http://backend_app;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```

Because no balancing method is explicitly set, NGINX uses round-robin by default.

# 8. Verification steps

## Verify security expectations

### From outside the internal network:

- `loadbalancer:8082` should be reachable
- `loadbalancer:80` should not be exposed
- `webserver1:80` and `webserver2:80` should not be reachable externally

### Test:

```bash
curl http://<LOADBALANCER_FLOATING_IP>:8082
```

Repeat several times:

```bash
for i in {1..6}; do curl -s http://<LOADBALANCER_FLOATING_IP>:8082 | grep "Hello from"; done
```

Expected output alternates between:

- `Hello from webserver1`
- `Hello from webserver2`

That demonstrates round-robin behavior.

---

## Verify load balancer is not exposing port 80

```bash
curl -I http://<LOADBALANCER_FLOATING_IP>:80
```

This should fail to connect.

---

## Verify web servers are internal-only

Since no floating IPs are attached to the web servers and their security group allows port 80 only from `subnet_cidr`, they are not directly accessible externally on port 80.

If you SSH to the load balancer, you can test internal access:

```bash
ssh -i ~/.ssh/id_rsa ubuntu@<LOADBALANCER_FLOATING_IP>
curl http://<WEB1_PRIVATE_IP>:80
curl http://<WEB2_PRIVATE_IP>:80
```

# 9. Notes on requirements mapping

- **Three VMs provisioned**: yes
- **Web SG**
  - SSH 22 from anywhere: yes
  - HTTP 80 only from internal network: yes, via `subnet_cidr`
- **LB SG**
  - SSH 22 from anywhere: yes
  - HTTP 8082 from external network: yes
- **Web servers not directly externally accessible on 80**:
  - no floating IPs
  - SG restricts access to internal CIDR only
- **Load balancer must not expose port 80**:
  - only port `lb_port` opened
  - nginx only listens on `lb_port`
- **Ports defined once using variables**:
  - Terraform: `web_port`, `lb_port`
  - passed into Ansible inventory and reused in templates/playbooks
- **Round-robin**:
  - default NGINX upstream behavior, no alternative method set
- **No hardcoded credentials**:
  - credentials are variables, ideally supplied via env vars or secure tfvars
