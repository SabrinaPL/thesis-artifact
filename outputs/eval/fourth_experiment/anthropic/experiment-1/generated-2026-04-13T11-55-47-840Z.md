### `ansible/playbooks/nfs_server.yml`

```yaml
---
- name: Configure NFS Server
  hosts: nfs_server
  become: true

  tasks:
    # ── Partition & format the attached volume ──────────────────
    - name: Ensure the attached volume has a filesystem
      community.general.filesystem:
        fstype: ext4
        dev: '{{ nfs_volume_device }}'

    - name: Create NFS mount point directory
      ansible.builtin.file:
        path: '{{ nfs_mount_point }}'
        state: directory
        mode: '0755'

    - name: Mount the attached volume
      ansible.posix.mount:
        path: '{{ nfs_mount_point }}'
        src: '{{ nfs_volume_device }}'
        fstype: ext4
        opts: defaults
        state: mounted

    - name: Create NFS export directory
      ansible.builtin.file:
        path: '{{ nfs_export_dir }}'
        state: directory
        owner: nobody
        group: nogroup
        mode: '0777'

    # ── Create default web content ──────────────────────────────
    - name: Create default index.html on shared storage
      ansible.builtin.copy:
        dest: '{{ nfs_export_dir }}/index.html'
        content: |
          <!DOCTYPE html>
          <html>
          <head><title>Shared NFS Content</title></head>
          <body>
            <h1>Hello from NFS-backed web application!</h1>
            <p>Served by: {{ '{{ ansible_hostname }}' }}</p>
          </body>
          </html>
        owner: nobody
        group: nogroup
        mode: '0644'
        force: false # Don't overwrite if already exists

    # ── Install and configure NFS ───────────────────────────────
    - name: Install NFS server packages
      ansible.builtin.apt:
        name:
          - nfs-kernel-server
          - nfs-common
        state: present
        update_cache: true

    - name: Configure NFS exports
      ansible.builtin.template:
        src: ../templates/exports.j2
        dest: /etc/exports
        owner: root
        group: root
        mode: '0644'
      notify: Restart NFS server

    - name: Ensure NFS server is enabled and running
      ansible.builtin.systemd:
        name: nfs-kernel-server
        state: started
        enabled: true

  handlers:
    - name: Restart NFS server
      ansible.builtin.systemd:
        name: nfs-kernel-server
        state: restarted
```

### `ansible/playbooks/web_servers.yml`

```yaml
---
- name: Configure Web Servers
  hosts: web_servers
  become: true

  vars:
    nfs_server_ip: "{{ hostvars[groups['nfs_server'][0]]['ansible_host'] }}"

  tasks:
    # ── Install required packages ───────────────────────────────
    - name: Install nginx and NFS client packages
      ansible.builtin.apt:
        name:
          - nginx
          - nfs-common
        state: present
        update_cache: true

    # ── Mount the NFS share ─────────────────────────────────────
    - name: Create web root mount point
      ansible.builtin.file:
        path: '{{ web_nfs_mount_point }}'
        state: directory
        mode: '0755'

    - name: Mount NFS shared directory
      ansible.posix.mount:
        path: '{{ web_nfs_mount_point }}'
        src: '{{ nfs_server_ip }}:{{ nfs_export_dir }}'
        fstype: nfs
        opts: 'rw,sync,hard,intr,timeo=30'
        state: mounted

    # ── Configure nginx ─────────────────────────────────────────
    - name: Remove default nginx site
      ansible.builtin.file:
        path: /etc/nginx/sites-enabled/default
        state: absent
      notify: Reload nginx

    - name: Deploy nginx virtual host configuration
      ansible.builtin.template:
        src: ../templates/nginx_web.conf.j2
        dest: /etc/nginx/sites-available/webapp.conf
        owner: root
        group: root
        mode: '0644'
      notify: Reload nginx

    - name: Enable the webapp site
      ansible.builtin.file:
        src: /etc/nginx/sites-available/webapp.conf
        dest: /etc/nginx/sites-enabled/webapp.conf
        state: link
      notify: Reload nginx

    - name: Ensure nginx is enabled and running
      ansible.builtin.systemd:
        name: nginx
        state: started
        enabled: true

  handlers:
    - name: Reload nginx
      ansible.builtin.systemd:
        name: nginx
        state: reloaded
```

        src: /etc/nginx/sites-available/loadbalancer.conf

      notify: Reload nginx

      ansible.builtin.systemd:
        name: nginx

## 4. Explanations

### Storage Architecture (Volume + NFS)

```
┌─────────────────────────────────────────────────────────────┐
│                   STORAGE ARCHITECTURE                       │
│                                                              │
│  ┌──────────────────────┐                                    │
│  │  Block Storage Volume │  Terraform-provisioned persistent │
│  │  (openstack_block-    │  volume (survives instance        │
│  │   storage_volume_v3)  │  termination).                    │
│  └──────────┬───────────┘                                    │
│             │ Attached via compute_volume_attach_v2           │
│  ┌──────────▼───────────┐                                    │
│  │     NFS Server        │                                   │
│  │  1. mkfs.ext4 /dev/vdb│  Ansible formats & mounts the    │
│  │  2. mount → /mnt/     │  volume, installs nfs-kernel-    │
│  │     nfs_data          │  server, and exports /mnt/       │
│  │  3. Export /mnt/      │  nfs_data/webroot to the         │
│  │     nfs_data/webroot  │  internal subnet.                │
│  └──────────┬───────────┘                                    │
│             │  NFS protocol (port 2049)                      │
│    ┌────────┴────────┐                                       │
│    │                  │                                       │
│  ┌─▼──────────┐  ┌──▼──────────┐                            │
│  │ Web Server 1│  │ Web Server 2│  NFS clients mount the     │
│  │ /var/www/   │  │ /var/www/   │  shared export onto their  │
│  │ html (NFS)  │  │ html (NFS)  │  document root.            │
│  └─────────────┘  └─────────────┘                            │
│                                                              │
│  BENEFITS:                                                   │
│  • Single source of truth for web content                    │
│  • Update once on NFS → both servers serve new content       │
│  • Block volume persists independently of NFS instance       │
│  • Volume can be snapshotted for backups                     │
└─────────────────────────────────────────────────────────────┘
```

The storage architecture separates **persistence** (block volume) from **sharing** (NFS). The block volume is a Terraform-managed resource that survives instance destruction — even if the NFS server is recreated, the volume and its data persist. NFS provides a POSIX-compliant shared filesystem that both web servers mount as their nginx document root. This means:

1. **Content consistency**: All web servers serve identical content automatically.
2. **Simplified updates**: Deploying new content requires writing to a single location.
3. **Data durability**: The block volume can be backed up, snapshotted, and reattached independently of any compute instance.

### CI/CD Workflow

```
┌──────────────┐     ┌──────────────┐     ┌──────────────┐     ┌──────────────┐
│   VALIDATE   │────▶│     PLAN     │────▶│    APPLY     │────▶│  CONFIGURE   │
│              │     │              │     │  (Manual ✋)  │     │  (Manual ✋)  │
│ • tf init    │     │ • tf plan    │     │ • tf apply   │     │ • Generate   │
│ • tf fmt     │     │ • Saves plan │     │ • Export     │     │   inventory  │
│ • tf validate│     │   artifact   │     │   outputs    │     │ • Run Ansible│
└──────────────┘     └──────────────┘     └──────────────┘     └──────────────┘
     ▲                                          │
     │  Runs on merge requests & main           │ Only on main branch
     └──────────────────────────────────────────┘
```

The pipeline implements a **progressive gate model**:

1. **Validate** (automatic): Runs on every merge request and commit to the default branch. Initializes Terraform with the remote backend, checks formatting (`terraform fmt -check`), and validates configuration syntax (`terraform validate`). Fast feedback loop for developers.

2. **Plan** (automatic): Generates an execution plan showing what Terraform will create, modify, or destroy. The plan is saved as an artifact so the exact plan reviewed is the one applied — no drift between plan and apply.

3. **Apply** (manual gate): Requires a human to click "Play" in GitLab after reviewing the plan output. This prevents accidental infrastructure changes. Exports Terraform outputs as JSON for the next stage.

4. **Configure** (manual gate): After infrastructure is provisioned, a human triggers Ansible configuration. The job dynamically generates an inventory file and variable overrides from Terraform outputs, then runs the master playbook in dependency order (NFS → Web → LB).

**Security measures in the pipeline:**

- No credentials stored in the repository; all secrets are GitLab CI/CD variables.
- `SSH_PRIVATE_KEY` is stored base64-encoded in a protected, masked CI/CD variable.
- `CI_JOB_TOKEN` is used for state backend authentication (ephemeral, scoped to the job).
- `TF_VAR_*` variables are injected from the CI/CD environment, never from committed files.

### Terraform State Management Approach

```
┌───────────────────────────────────────────────────────┐
│              GITLAB-MANAGED TERRAFORM STATE            │
│                                                        │
│  Developer ──push──▶ GitLab CI ──HTTP backend──▶ State │
│                         │                        Store │
│                         │  Uses CI_JOB_TOKEN           │
│                         │  for authentication           │
│                         │                               │
│                    ┌────▼────┐                          │
│                    │  State  │  Per-environment:        │
│                    │  Lock   │  webapp-dev              │
│                    └─────────┘  webapp-staging          │
│                                 webapp-production       │
└───────────────────────────────────────────────────────┘
```

The solution uses **GitLab's built-in HTTP backend** for Terraform state management:

- **Remote storage**: State is stored in GitLab's infrastructure, not locally or in the repository. This prevents state file conflicts and accidental exposure of sensitive data that Terraform stores in state (IPs, resource IDs, etc.).

- **State locking**: The HTTP backend supports `lock_address` and `unlock_address`, preventing concurrent modifications. If two pipelines attempt to run `terraform apply` simultaneously, the second will wait or fail rather than corrupting state.

- **Authentication**: Uses `CI_JOB_TOKEN`, a short-lived token automatically injected by GitLab into each CI job. No long-lived credentials need to be managed for state access.

- **Per-environment state**: The state name includes the environment (`webapp-dev`, `webapp-staging`, `webapp-production`), providing complete isolation between environments. Each environment has its own state file and lock.

- **No secrets in VCS**: The backend is configured entirely via `-backend-config` CLI flags that reference CI/CD variables, following the principle that the `backend "http" {}` block in code is empty — all sensitive configuration is injected at runtime.
