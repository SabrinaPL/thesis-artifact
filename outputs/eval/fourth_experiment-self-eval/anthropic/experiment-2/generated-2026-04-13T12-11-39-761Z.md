### `terraform/backend.tf`

```hcl
# ============================================================================
# REMOTE BACKEND CONFIGURATION — Where Terraform Stores Its "Memory"
# ============================================================================
#
# WHAT IS TERRAFORM STATE?
# When Terraform creates infrastructure, it records what it created in a
# "state file" (terraform.tfstate). This file is Terraform's memory — it maps
# your HCL code to real cloud resources. Without it, Terraform wouldn't know
# what already exists and would try to create everything from scratch.
#
# WHY REMOTE STATE?
# By default, state is stored locally in a file. This is problematic because:
#   1. If your laptop dies, you lose track of all your infrastructure.
#   2. Two people can't work on the same infrastructure simultaneously.
#   3. CI/CD runners are ephemeral — state would be lost between pipeline runs.
#
# A remote backend solves all three problems by storing state in a shared,
# persistent, encrypted location.
#
# STATE LOCKING:
# The DynamoDB table provides a "lock" — like a mutex in programming.
# When one person/pipeline runs `terraform apply`, it acquires the lock.
# If someone else tries to run simultaneously, they'll get a "state locked"
# error instead of corrupting the state. This is CRITICAL for team safety.
#
# CONFIGURATION APPROACH:
# We use partial configuration here. The bucket, key, and table are defined
# because they're structural (not secret). However, in CI/CD, you can also
# pass these via -backend-config flags for maximum flexibility:
#   terraform init \
#     -backend-config="bucket=$TF_STATE_BUCKET" \
#     -backend-config="key=$TF_STATE_KEY"
# ============================================================================

terraform {
  backend "s3" {
    # These values can be overridden using -backend-config in CI/CD.
    # They are NOT secrets — they describe WHERE state lives, not HOW to access it.
    bucket         = "terraform-state-bucket"
    key            = "load-balanced-web-app/terraform.tfstate"
    region         = "us-east-1"

    # Encrypt the state file at rest. State files often contain sensitive data
    # like IP addresses, resource IDs, and sometimes even passwords embedded by
    # providers. Encryption ensures this data is protected in storage.
    encrypt = true

    # DynamoDB table used for state locking (prevents concurrent modifications).
    dynamodb_table = "terraform-state-lock"

    # Authentication credentials (AWS_ACCESS_KEY_ID, AWS_SECRET_ACCESS_KEY)
    # are injected via environment variables in CI/CD — never written here.
  }
}
```

```hcl
# ============================================================================
# NETWORKING — The Virtual "Wiring" of Our Infrastructure
# ============================================================================
#
# ARCHITECTURE OVERVIEW:
# We create a private network (like a home LAN) for our servers to
# communicate securely. Only the load balancer gets a "floating IP"
# (public IP address) so external users can reach the application.
#
#   Internet ──→ Floating IP ──→ Load Balancer ──→ Internal Network
#                                                    ├── Web Server 1
#                                                    ├── Web Server 2
#                                                    └── NFS Server
#
# WHY PRIVATE NETWORKING?
# Placing servers on a private network reduces the "attack surface."
# If web servers and NFS servers don't have public IPs, attackers can't
# directly reach them from the internet — they'd have to go through the
# load balancer first.
# ============================================================================

# ──────────────────────────────────────────────
# Data Source: Look up the external network
# ──────────────────────────────────────────────
# A "data source" reads information about something that already exists
# (as opposed to a "resource" which creates something new).
# The external network is managed by the OpenStack administrator —
# we just need to reference it.
data "openstack_networking_network_v2" "external" {
  name = var.external_network_name
}

# ──────────────────────────────────────────────
# Internal Network + Subnet
# ──────────────────────────────────────────────
# Think of a "network" as the physical cable, and "subnet" as the
# IP address scheme used on that cable. You need both.
resource "openstack_networking_network_v2" "internal" {
  name           = "${var.project_name}-${var.environment}-network"
  admin_state_up = true

  # TAGGING BEST PRACTICE: Tags help you identify resources in the cloud
  # console and are essential for cost tracking and automated governance.
  tags = [var.project_name, var.environment, "managed-by-terraform"]
}

resource "openstack_networking_subnet_v2" "internal" {
  name            = "${var.project_name}-${var.environment}-subnet"
  network_id      = openstack_networking_network_v2.internal.id
  cidr            = var.network_cidr
  ip_version      = 4
  dns_nameservers = var.dns_nameservers

  tags = [var.project_name, var.environment]
}

# ──────────────────────────────────────────────
# Router: Connects internal network to the internet
# ──────────────────────────────────────────────
# Without a router, instances on the internal network can talk to each other
# but can't reach the internet (for package installs, etc.). The router
# acts as a gateway, performing NAT (Network Address Translation) for
# outbound traffic.
resource "openstack_networking_router_v2" "router" {
  name                = "${var.project_name}-${var.environment}-router"
  external_network_id = data.openstack_networking_network_v2.external.id

  tags = [var.project_name, var.environment]
}

# Connect the router to our internal subnet.
# This is like plugging a cable from the router into the internal network switch.
resource "openstack_networking_router_interface_v2" "router_iface" {
  router_id = openstack_networking_router_v2.router.id
  subnet_id = openstack_networking_subnet_v2.internal.id
}

# ──────────────────────────────────────────────
# Floating IP for the Load Balancer
# ──────────────────────────────────────────────
# A "floating IP" is a public IP address that can be attached to (and
# detached from) instances. Only the load balancer needs one because
# it's the only entry point for external users.
resource "openstack_networking_floatingip_v2" "lb" {
  pool        = var.external_network_name
  description = "Public IP for ${var.project_name} load balancer"
}

resource "openstack_compute_floatingip_associate_v2" "lb" {
  floating_ip = openstack_networking_floatingip_v2.lb.address
  instance_id = openstack_compute_instance_v2.load_balancer.id
}
```

### `terraform/security_groups.tf`

```hcl
# ============================================================================
# SECURITY GROUPS — The Firewall Rules for Each Server Role
# ============================================================================
#
# WHAT ARE SECURITY GROUPS?
# Security groups are virtual firewalls that control inbound (ingress) and
# outbound (egress) network traffic to instances. Each instance can belong
# to one or more security groups.
#
# PRINCIPLE OF LEAST PRIVILEGE:
# Each server role gets its own security group with ONLY the ports it needs.
# This means if an attacker compromises a web server, they can't automatically
# access NFS ports because the web server's security group doesn't allow
# outbound NFS traffic initiation to arbitrary hosts (though in this case
# the NFS server explicitly trusts the web security group).
#
# RULE OF THUMB: Start with "deny all" and only open what's necessary.
# OpenStack security groups deny all ingress by default, which is good.
# ============================================================================

# ──────────────────────────────────────────────
# Security Group: Web Servers
# ──────────────────────────────────────────────
resource "openstack_networking_secgroup_v2" "web" {
  name        = "${var.project_name}-${var.environment}-web-sg"
  description = "Allow SSH from trusted CIDR and HTTP from internal network only"
}

# SSH access — restricted to the CIDR range you specify.
# SECURITY NOTE: In production, this should be your VPN or bastion host IP.
# If you use "0.0.0.0/0", bots will find your server within minutes and
# start brute-force attacks.
resource "openstack_networking_secgroup_rule_v2" "web_ssh" {
  security_group_id = openstack_networking_secgroup_v2.web.id
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = var.ssh_allowed_cidr
  description       = "SSH access from trusted CIDR"
}

# HTTP access — only from the internal network.
# Web servers should NOT be directly accessible from the internet.
# External traffic must go through the load balancer, which forwards
# requests to web servers on port 80 within the private network.
resource "openstack_networking_secgroup_rule_v2" "web_http_internal" {
  security_group_id = openstack_networking_secgroup_v2.web.id
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = var.network_cidr
  description       = "HTTP from internal network only (via load balancer)"
}

# Security Group: Load Balancer
# ──────────────────────────────────────────────

}

  protocol          = "tcp"

  port_range_max    = 22

  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 2049
  port_range_max    = 2049
  remote_group_id   = openstack_networking_secgroup_v2.web.id

}

  port_range_min    = 111

  description       = "rpcbind TCP for NFS service discovery from web servers"
}

  security_group_id = openstack_networking_secgroup_v2.nfs.id
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "udp"
  port_range_min    = 111
```

```hcl
# BLOCK STORAGE — Persistent Data Volume for NFS
# ============================================================================
#
# WHY SEPARATE STORAGE FROM COMPUTE?
# Compute instances (VMs) have "ephemeral" storage — when you delete the VM,
# the storage is gone. Block storage volumes are INDEPENDENT of VMs.
# Think of it like an external hard drive you can unplug from one computer
# and plug into another.
#
# In our architecture:
#   - If the NFS server VM crashes, we can create a new VM and re-attach
#     the volume — all data is preserved.
#   - The volume can be backed up (snapshotted) independently.
#   - The volume size can be expanded later without replacing the VM.
#
# This is the foundation of our data durability strategy.
# ============================================================================

resource "openstack_blockstorage_volume_v3" "nfs_data" {
  name        = "${var.project_name}-${var.environment}-nfs-data"
  description = "Persistent block storage for NFS shared data. This volume survives VM deletion."
  size        = var.nfs_volume_size

  # METADATA acts like tags for block storage. Useful for identifying
  # volumes in the OpenStack dashboard and for automation scripts.
  metadata = {
    project     = var.project_name
    environment = var.environment
    role        = "nfs-data"
    managed_by  = "terraform"
  }
}

# Attach the volume to the NFS server instance.
# After attachment, the volume appears as a block device (like /dev/vdb)
# inside the VM. Ansible will then format it and mount it.
```

### `ansible/playbooks/nfs_server.yml`

```yaml
---
# ============================================================================
# PLAYBOOK: Configure NFS Server
# ============================================================================
#
# WHAT THIS PLAYBOOK DOES:
# 1. Formats the attached block storage volume with a filesystem
# 2. Mounts the volume to a directory
# 3. Installs NFS server software
# 4. Exports the directory so web servers can mount it over the network
#
# WHY NFS?
# NFS (Network File System) allows multiple servers to share the same files
# over the network as if they were local. This means:
#   - Both web servers serve the exact same content
#   - Content updates only need to happen in ONE place
#   - No need for file synchronization tools (rsync, etc.)
# ============================================================================

- name: Configure NFS Server
  hosts: nfs_servers
  become: true  # Run as root (needed for package installs, mounting, etc.)

  vars:
    # Where the volume will be mounted inside the NFS server
    nfs_mount_point: /srv/nfs/shared

    # The block device path — comes from Terraform output via inventory
    nfs_volume_device: "{{ hostvars[inventory_hostname]['nfs_volume_device'] | default('/dev/vdb') }}"

    # ext4 is a reliable, well-tested Linux filesystem
    nfs_filesystem_type: ext4

    # SECURITY: Only allow NFS access from our internal network CIDR.
    # This prevents any host outside our network from mounting the share.
    nfs_export_network: "{{ lookup('env', 'NETWORK_CIDR') | default('10.0.1.0/24', true) }}"

    # NFS export options explained:
    #   rw              → Allow read and write access
    #   sync            → Write data to disk before confirming to clients (safer, slightly slower)
    #   no_subtree_check → Improves reliability when exporting subdirectories
    #   root_squash     → SECURITY: Maps remote root user to 'nobody'. This prevents
    #                     a compromised web server from having root access to NFS files.
    #                     The original code used no_root_squash which is a security risk!
    nfs_export_options: "rw,sync,no_subtree_check,root_squash"

  tasks:
    - name: Wait for system to be ready
      wait_for_connection:
        timeout: 120

    - name: Install NFS server packages
      apt:
        name:
          - nfs-kernel-server
          - nfs-common
        state: present
        update_cache: true
        cache_valid_time: 3600  # Don't re-update apt cache if updated within the last hour

    - name: Check if volume is already formatted
      command: blkid {{ nfs_volume_device }}
      register: blkid_result
      failed_when: false  # Don't fail if the device has no filesystem yet
      changed_when: false  # This is a read-only check, not a change

    - name: Create filesystem on the attached volume
      filesystem:
        fstype: "{{ nfs_filesystem_type }}"
        dev: "{{ nfs_volume_device }}"
      when: blkid_result.rc != 0  # Only format if no filesystem exists

    - name: Create the NFS mount point directory
      file:
        path: "{{ nfs_mount_point }}"
        state: directory
        owner: nobody
        group: nogroup
        mode: "0755"

    - name: Mount the block storage volume persistently
      mount:
        path: "{{ nfs_mount_point }}"
        src: "{{ nfs_volume_device }}"
        fstype: "{{ nfs_filesystem_type }}"
        opts: defaults,nofail
        # 'nofail' prevents the system from hanging on boot if the volume
        # is temporarily unavailable. Important for cloud environments.
        state: mounted
        # 'mounted' ensures the entry is in /etc/fstab (survives reboots)
        # AND the volume is currently mounted.

    - name: Set ownership and permissions on mounted directory
      file:
        path: "{{ nfs_mount_point }}"
        owner: nobody
        group: nogroup
        mode: "0755"
        state: directory

    - name: Create default index page for web content
      copy:
        dest: "{{ nfs_mount_point }}/index.html"
        content: |
          <!DOCTYPE html>
          <html>
          <head><title>Shared NFS Web Content</title></head>
          <body>
            <h1>Hello from NFS-backed web application!</h1>
            <p>Served from shared persistent storage.</p>
            <p>Both web servers serve this same file from NFS.</p>
          </body>
          </html>
        owner: www-data
        group: www-data
        mode: "0644"
        force: false  # Don't overwrite if file already exists (preserves content updates)

    - name: Configure NFS exports
      template:
        src: ../templates/exports.j2
        dest: /etc/exports
        owner: root
        group: root
        mode: "0644"
        # 'validate' runs a syntax check before deploying the file.
        # If validation fails, the old file is kept — preventing NFS breakage.
        validate: 'exportfs -r && echo valid || echo invalid'
      notify: Restart NFS server

    - name: Apply NFS exports immediately
      command: exportfs -ra
      changed_when: false

    - name: Ensure NFS server is running and enabled on boot
      systemd:
        name: nfs-kernel-server
        state: started
        enabled: true

  handlers:
    # Handlers run only when "notified" by a task that made a change.
    # This avoids restarting NFS unnecessarily on every playbook run.
    - name: Restart NFS server
      systemd:
        name: nfs-kernel-server
        state: restarted
```

```yaml
# ============================================================================
# PLAYBOOK: Configure Web Servers
# ============================================================================
#
# WHAT THIS PLAYBOOK DOES:
# 1. Installs nginx (web server software)
# 2. Mounts the NFS shared directory from the NFS server
# 3. Configures nginx to serve content from the NFS mount
#
# IMPORTANT: The NFS server MUST be configured before running this playbook!
# The CI/CD pipeline enforces this ordering.
# ============================================================================

- name: Configure Web Servers
  hosts: web_servers
  become: true

  vars:
    # IP address of the NFS server (populated from Terraform outputs)
    nfs_server_ip: "{{ lookup('env', 'NFS_SERVER_IP') }}"

    # Remote path on the NFS server that we're mounting
    nfs_remote_path: /srv/nfs/shared

    # Local path where the NFS share will be accessible on this web server
    nfs_local_mount: /var/www/shared

    # Nginx will serve files from this directory
    nginx_document_root: /var/www/shared

  tasks:
    - name: Wait for system to be ready
      wait_for_connection:
        timeout: 120

    - name: Install required packages
      apt:
        name:
          - nginx
          - nfs-common  # NFS client tools (needed to mount NFS shares)
        state: present
        update_cache: true
        cache_valid_time: 3600

    - name: Create local NFS mount point directory
      file:
        path: "{{ nfs_local_mount }}"
        state: directory
        owner: www-data   # nginx runs as www-data user
        group: www-data
        mode: "0755"

    - name: Wait for NFS server to be reachable
      wait_for:
        host: "{{ nfs_server_ip }}"
        port: 2049
        timeout: 120
        state: started
      # This prevents the mount task from failing if the NFS server
      # hasn't finished starting up yet.

    - name: Mount NFS shared directory
      mount:
        path: "{{ nfs_local_mount }}"
        src: "{{ nfs_server_ip }}:{{ nfs_remote_path }}"
        fstype: nfs
        opts: "defaults,_netdev,nofail,soft,timeo=150,retrans=3"
        # Mount options explained:
        #   _netdev   → Wait for network before attempting mount (essential for NFS)
        #   nofail    → Don't block system boot if NFS is temporarily unavailable
        #   soft      → Return an error after retrans retries instead of hanging forever
        #              (hard mounts can cause processes to hang indefinitely if NFS goes down)
        #   timeo=150 → Wait 15 seconds (in deciseconds) between retries
        #   retrans=3 → Retry 3 times before giving up
        state: mounted

    - name: Deploy nginx virtual host configuration
      template:
        src: ../templates/nginx_web.conf.j2
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: "0644"
      notify: Reload nginx

    - name: Remove default nginx welcome page if it exists
      file:
        path: /var/www/html/index.nginx-debian.html
        state: absent

    - name: Ensure nginx is running and enabled on boot
      systemd:
        name: nginx
        state: started
        enabled: true

  handlers:
    - name: Reload nginx
      systemd:
        name: nginx
        state: reloaded
```

```yaml
    #   map('extract', ...)       → Get the ansible_host variable from each
    #   list                      → Convert the result to a list
    web_server_ips: "{{ groups['web_servers'] | map('extract', hostvars, ['ansible_host']) | list }}"
```

```jinja2
# ============================================================================
# Nginx Load Balancer Configuration
# Managed by Ansible — do not edit manually!
# ============================================================================
#
# HOW LOAD BALANCING WORKS:
# The "upstream" block defines a pool of backend servers. When a request
# arrives on port {{ lb_listen_port }}, nginx forwards it to one of the
# servers in the pool.
#
# ROUND-ROBIN (default method):
# Requests are distributed to servers in sequential order:
#   Request 1 → Web Server 1
#   Request 2 → Web Server 2
#   Request 3 → Web Server 1
#   Request 4 → Web Server 2
#   ... and so on.
#
# This ensures even distribution of load across all web servers.

upstream web_backend {
    # Each 'server' line adds a backend to the pool.
    # These IPs are injected dynamically from the Ansible inventory.
{% for ip in web_server_ips %}
    server {{ ip }}:{{ web_server_port }} max_fails=3 fail_timeout=30s;
    # max_fails=3    → After 3 consecutive failures, mark server as "down"
    # fail_timeout=30s → Keep server marked as "down" for 30 seconds before retrying
    # This provides basic health checking — if a web server crashes,
    # the load balancer stops sending traffic to it automatically.
{% endfor %}
}

server {
    # Listen on port {{ lb_listen_port }} for incoming HTTP requests.
    listen {{ lb_listen_port }} default_server;
    listen [::]:{{ lb_listen_port }} default_server;

    server_name _;

    # Request/response size limits and timeouts to protect against
    # slow clients and oversized requests.
    client_max_body_size 10m;

    location / {
        # Forward all requests to the upstream pool defined above.
        proxy_pass http://web_backend;

        # These headers preserve the original client's information so the
        # web servers know who the real client is (not just the load balancer).
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Timeout settings prevent the load balancer from waiting forever
        # if a web server is unresponsive.
        proxy_connect_timeout 10s;   # Time to establish connection to backend
        proxy_read_timeout    30s;   # Time to wait for backend to send response
        proxy_send_timeout    10s;   # Time to wait for backend to accept data

        # If a backend fails, try the next one in the upstream pool.
        proxy_next_upstream error timeout http_502 http_503 http_504;
        proxy_next_upstream_tries 2;
    }

    # Load balancer health check endpoint.
    # Useful for monitoring systems to verify the LB itself is alive.
    location /health {
        access_log off;
        return 200 "Load Balancer OK\n";
        add_header Content-Type text/plain;
    }
}
```

```bash
#!/usr/bin/env bash
# ============================================================================
# SCRIPT: Generate Ansible Inventory from Terraform Outputs
# ============================================================================
#
# This script bridges Terraform and Ansible by:
# 1. Reading Terraform output values (IP addresses, device paths)
# 2. Exporting them as environment variables that the Ansible inventory uses
#
# USAGE:
#   source scripts/generate_inventory.sh tf_outputs.json
#
# The 'source' command is important — it exports variables into the current
# shell session so Ansible can read them.
# ============================================================================

set -euo pipefail

# Check if jq is installed (jq is a command-line JSON processor)
if ! command -v jq &> /dev/null; then
    echo "ERROR: jq is required but not installed. Install with: apt-get install jq"
    exit 1
fi

TF_OUTPUT_FILE="${1:-tf_outputs.json}"

if [ ! -f "${TF_OUTPUT_FILE}" ]; then
    echo "ERROR: Terraform output file not found: ${TF_OUTPUT_FILE}"
    echo "Run: cd terraform && terraform output -json > ../tf_outputs.json"
    exit 1
fi

echo "Parsing Terraform outputs from ${TF_OUTPUT_FILE}..."

export NFS_SERVER_IP=$(jq -r '.nfs_server_ip.value' "${TF_OUTPUT_FILE}")
export NFS_VOLUME_DEVICE=$(jq -r '.nfs_volume_device.value' "${TF_OUTPUT_FILE}")
export WEB_SERVER_1_IP=$(jq -r '.web_server_ips.value[0]' "${TF_OUTPUT_FILE}")
export WEB_SERVER_2_IP=$(jq -r '.web_server_ips.value[1]' "${TF_OUTPUT_FILE}")
export LB_FLOATING_IP=$(jq -r '.load_balancer_floating_ip.value' "${TF_OUTPUT_FILE}")
export NETWORK_CIDR=$(jq -r '.network_cidr.value' "${TF_OUTPUT_FILE}")

echo "Exported inventory variables:"
echo "  NFS_SERVER_IP     = ${NFS_SERVER_IP}"
echo "  NFS_VOLUME_DEVICE = ${NFS_VOLUME_DEVICE}"
echo "  WEB_SERVER_1_IP   = ${WEB_SERVER_1_IP}"
echo "  WEB_SERVER_2_IP   = ${WEB_SERVER_2_IP}"
echo "  LB_FLOATING_IP    = ${LB_FLOATING_IP}"
echo "  NETWORK_CIDR      = ${NETWORK_CIDR}"
```
