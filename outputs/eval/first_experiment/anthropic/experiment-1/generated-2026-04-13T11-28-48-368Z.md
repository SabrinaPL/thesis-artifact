### `terraform/main.tf`

```hcl
terraform {
  required_version = ">= 1.3.0"

  required_providers {
    openstack = {
      source  = "terraform-provider-openstack/openstack"
      version = "~> 1.54"
    }
  }
}

# Provider configuration — credentials are sourced from environment variables
# (OS_AUTH_URL, OS_TENANT_NAME, OS_USERNAME, OS_PASSWORD, etc.)
provider "openstack" {
  region = var.region
}

# -------------------------------------------------------------------
# Security Group
# -------------------------------------------------------------------
resource "openstack_networking_secgroup_v2" "web_sg" {
  name                 = var.security_group_name
  description          = "Security group for web server: allows SSH and HTTP inbound"
  delete_default_rules = true
}

# Default egress — IPv4
resource "openstack_networking_secgroup_rule_v2" "egress_ipv4" {
  direction         = "egress"
  ethertype         = "IPv4"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# Default egress — IPv6
resource "openstack_networking_secgroup_rule_v2" "egress_ipv6" {
  direction         = "egress"
  ethertype         = "IPv6"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# Ingress — SSH (port 22)
resource "openstack_networking_secgroup_rule_v2" "ingress_ssh" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 22
  port_range_max    = 22
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# Ingress — HTTP (port 80)
resource "openstack_networking_secgroup_rule_v2" "ingress_http" {
  direction         = "ingress"
  ethertype         = "IPv4"
  protocol          = "tcp"
  port_range_min    = 80
  port_range_max    = 80
  remote_ip_prefix  = "0.0.0.0/0"
  security_group_id = openstack_networking_secgroup_v2.web_sg.id
}

# -------------------------------------------------------------------
# Floating IP
# -------------------------------------------------------------------
resource "openstack_networking_floatingip_v2" "web_fip" {
  pool = var.floating_ip_pool
}

# -------------------------------------------------------------------
# Compute Instance
# -------------------------------------------------------------------
resource "openstack_compute_instance_v2" "web_server" {
  name              = var.instance_name
  image_name        = var.image_name
  flavor_name       = var.flavor_name
  key_pair          = var.ssh_key_name
  availability_zone = var.availability_zone

  security_groups = [
    openstack_networking_secgroup_v2.web_sg.name
  ]

  network {
    name = var.network_name
  }
}

# -------------------------------------------------------------------
# Associate Floating IP with Instance
# -------------------------------------------------------------------
resource "openstack_compute_floatingip_associate_v2" "web_fip_assoc" {
  floating_ip = openstack_networking_floatingip_v2.web_fip.address
  instance_id = openstack_compute_instance_v2.web_server.id
}

# -------------------------------------------------------------------
# Generate Ansible Inventory
# -------------------------------------------------------------------
resource "local_file" "ansible_inventory" {
  content = templatefile("${path.module}/../ansible/inventory.tpl", {
    server_ip        = openstack_networking_floatingip_v2.web_fip.address
    ssh_user         = var.ssh_user
    private_key_path = var.private_key_path
  })
  filename = "${path.module}/../ansible/inventory.ini"

  depends_on = [openstack_compute_floatingip_associate_v2.web_fip_assoc]
}
```

### `ansible/playbook.yml`

```yaml
---
- name: Configure web server with nginx
  hosts: webservers
  become: true
  gather_facts: true

  vars:
    nginx_document_root: /var/www/html
    nginx_index_file: index.html
    nginx_listen_port: 80

  tasks:
    # ------------------------------------------------------------------
    # Package installation
    # ------------------------------------------------------------------
    - name: Update apt cache (Debian/Ubuntu)
      ansible.builtin.apt:
        update_cache: yes
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"

    - name: Install nginx (Debian/Ubuntu)
      ansible.builtin.apt:
        name: nginx
        state: present
      when: ansible_os_family == "Debian"

    - name: Install nginx (RedHat/CentOS)
      ansible.builtin.yum:
        name: nginx
        state: present
      when: ansible_os_family == "RedHat"

    # ------------------------------------------------------------------
    # Ensure document root exists
    # ------------------------------------------------------------------
    - name: Create document root directory
      ansible.builtin.file:
        path: "{{ nginx_document_root }}"
        state: directory
        owner: root
        group: root
        mode: "0755"

    # ------------------------------------------------------------------
    # Deploy static HTML page
    # ------------------------------------------------------------------
    - name: Deploy static index.html
      ansible.builtin.copy:
        src: files/index.html
        dest: "{{ nginx_document_root }}/{{ nginx_index_file }}"
        owner: root
        group: root
        mode: "0644"
      notify: Reload nginx

    # ------------------------------------------------------------------
    # Configure nginx default site
    # ------------------------------------------------------------------
    - name: Deploy nginx default site configuration
      ansible.builtin.copy:
        dest: /etc/nginx/sites-available/default
        owner: root
        group: root
        mode: "0644"
        content: |
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              root {{ nginx_document_root }};
              index {{ nginx_index_file }};

              server_name _;

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      when: ansible_os_family == "Debian"
      notify: Reload nginx

    - name: Deploy nginx configuration (RedHat)
      ansible.builtin.copy:
        dest: /etc/nginx/conf.d/default.conf
        owner: root
        group: root
        mode: "0644"
        content: |
          server {
              listen {{ nginx_listen_port }} default_server;
              listen [::]:{{ nginx_listen_port }} default_server;

              root {{ nginx_document_root }};
              index {{ nginx_index_file }};

              server_name _;

              location / {
                  try_files $uri $uri/ =404;
              }
          }
      when: ansible_os_family == "RedHat"
      notify: Reload nginx

    # ------------------------------------------------------------------
    # Ensure nginx is enabled and running
    # ------------------------------------------------------------------
    - name: Enable and start nginx service
      ansible.builtin.systemd:
        name: nginx
        enabled: true
        state: started

  handlers:
    - name: Reload nginx
      ansible.builtin.systemd:
        name: nginx
        state: reloaded
```

## Deployment Instructions

### Prerequisites

| Tool | Minimum Version |
|------|-----------------|
| Terraform | >= 1.3 |
| Ansible | >= 2.12 |
| OpenStack CLI (optional) | any |

### Step 1: Set OpenStack Credentials

Export your OpenStack credentials as environment variables. You can source your `openrc` file downloaded from the OpenStack dashboard:

```bash
source ~/openrc.sh

# Or set variables manually:
export OS_AUTH_URL="https://your-openstack:5000/v3"
export OS_PROJECT_NAME="my-project"
export OS_USER_DOMAIN_NAME="Default"
export OS_PROJECT_DOMAIN_NAME="Default"
export OS_USERNAME="my-username"
export OS_PASSWORD="my-password"
export OS_REGION_NAME="RegionOne"
export OS_IDENTITY_API_VERSION=3
```

### Step 2: Configure Terraform Variables

```bash
cd terraform/
cp terraform.tfvars.example terraform.tfvars
```

Edit `terraform.tfvars` with your environment-specific values:

```hcl
ssh_key_name      = "my-keypair"
private_key_path  = "~/.ssh/id_rsa"
image_name        = "Ubuntu 22.04"
flavor_name       = "m1.small"
network_name      = "private"
floating_ip_pool  = "public"
ssh_user          = "ubuntu"
```

> **Note:** The `ssh_key_name` must match a keypair already registered in OpenStack. If you haven't created one, run:
> ```bash
> openstack keypair create --public-key ~/.ssh/id_rsa.pub my-keypair
> ```

### Step 3: Provision Infrastructure with Terraform

```bash
cd terraform/

# Initialize the provider
terraform init

# Review the execution plan
terraform plan

# Apply the configuration
terraform apply
```

After successful apply, Terraform will output:
- The floating IP address
- The web URL
- The SSH command
- An auto-generated Ansible inventory file at `ansible/inventory.ini`

### Step 4: Wait for Instance to Boot

Allow 30–60 seconds for the instance to fully boot and become SSH-accessible:

```bash
# Test SSH connectivity (use the IP from Terraform output)
FLOATING_IP=$(terraform output -raw floating_ip)
ssh -i ~/.ssh/id_rsa -o StrictHostKeyChecking=no ubuntu@${FLOATING_IP} "echo 'SSH is ready'"
```

### Step 5: Configure the Server with Ansible

```bash
cd ../ansible/

# Run the playbook using the auto-generated inventory
ansible-playbook -i inventory.ini playbook.yml
```

Expected output should show all tasks completing with `ok` or `changed` status.

### Step 6: Verify HTTP Access

**Using a browser:**
```
Open http://<FLOATING_IP> in your browser
```

**Using curl:**
```bash
FLOATING_IP=$(cd ../terraform && terraform output -raw floating_ip)
curl http://${FLOATING_IP}
```

You should see the HTML page with **"Hello from OpenStack!"**.

**Using the Terraform output directly:**
```bash
cd ../terraform
echo "Visit: $(terraform output -raw web_url)"
```

### Teardown

To destroy all provisioned resources:

```bash
cd terraform/
terraform destroy
```
